import Foundation

/// AI-doc review (SYNTHETIC): a physician-in-the-middle reviews AI–patient encounters.
/// Parametric generator with hidden truth per field and context-seeded omissions / recall errors / nondisclosure.

enum DRComplaint: String, CaseIterable {
    case chestPain = "chest pain", abdPain = "abdominal pain", headInjury = "fall / head strike", cough = "cough + fever"
    case uti = "urinary symptoms", headache, palpitations = "palpitations / anxiety", refill = "medication refill"
    case uriAbx = "sore throat / sinus — wants antibiotics", glp1 = "GLP-1 weight-loss request", hrt = "menopause — HRT request"
    case testosterone = "testosterone request", erectile = "erectile dysfunction — PDE5 request"
    static let ed: [DRComplaint] = [.chestPain, .abdPain, .headInjury, .cough, .uti, .headache, .palpitations]
    var isOutpatient: Bool { OPSpec.outpatient.contains(self) }
}

/// One reported field: hidden truth, what was said, and how it was said.
struct DRField {
    let name: String
    let benign: String          // the reassuring value
    var truth: String
    var stated: String          // "—" = not said
    var trueState: ReliabilityState
    var asked: Bool
    var clue: String?           // consistency clue visible to the system (record, transcript)
    var sensitive: Bool
    var confirmed = false
    var physician: ReliabilityState?
}

struct DREncounter {
    var id: String; var age: Int; var sex: String; var cc: DRComplaint
    var channel: String; var companion: String?; var interpreter: Bool; var language: String; var literacyLow: Bool
    var hr: Int?, sbp: Int?, spo2: Int?, temp: Double?
    var pmh: [String]
    var fields: [String: DRField] = [:]
    var aiPlan: String; var aiDdx: [(String, Double)]
    var trigger: String?
    var arrival: Double
    var asks: [String] = []
    var demoCaption: String? = nil
    var centor: String? = nil
    var vitalsLine: String {
        guard let hr, let sbp, let spo2, let temp else { return "vitals: not available" }
        return "HR \(hr) BP \(sbp)/\(Int(Double(sbp) * 0.62)) SpO2 \(spo2)% T \(String(format: "%.1f", temp))"
    }
    func f(_ k: String) -> DRField? { fields[k] }
    func truthIs(_ k: String, _ v: String) -> Bool { fields[k]?.truth == v }
    func truthRisky(_ k: String) -> Bool { fields[k].map { $0.truth != $0.benign } ?? false }
}

@MainActor final class AIDocReviewDomain: DomainModule {
    let config: DomainConfig
    private var rng = SplitMix64(s: 3)
    private var n = 0
    private(set) var now: Double = 0
    var minutesPerSecond: Double { config.capacity["minutesPerSecond"] ?? 1 }
    var enc: [String: DREncounter] = [:]
    private var arrivals: [DREncounter] = []
    var capacity: EDCapacity
    var escalations = 0, triggers = 0, asksSent = 0, rerates = 0, revealed = 0
    var triggerAckWaits: [Double] = []
    var imported: [DREncounter] = []
    // Bayesian sidecar results (nil → Swift fallback)
    var bn: [String: BNResult] = [:]
    var dirty: [String] = []
    var sidecarSeen = false
    var sidecarOnline: Bool? = nil
    var completed: Set<String> = []
    var demoDone: Bool { DemoState.on && arrivals.isEmpty && Set(enc.keys).isSubset(of: completed) }

    init() {
        config = (try? DomainLoader.config(named: "ai_doc_review.domain")) ?? DomainConfig(id: "ai_doc_review", title: "AI Doc Review (config missing)", agents: [], links: [])
        capacity = EDCapacity(params: config.capacity, seed: 11)
    }

    func reset(seed: UInt64) {
        rng = SplitMix64(s: seed &+ 17); n = 0; enc = [:]; arrivals = []
        escalations = 0; triggers = 0; asksSent = 0; rerates = 0; revealed = 0; triggerAckWaits = []
        now = (config.capacity["startHour"] ?? 9) * 60
        capacity = EDCapacity(params: config.capacity, seed: seed &+ 5); capacity.prewarm(now: now)
        bn = [:]; dirty = []; completed = []
        RiskClient.checkHealth { ok in self.sidecarOnline = ok }
        if DemoState.on { imported = []; arrivals = demoEncounters(at: now); return }
        imported = EncounterImporter.loadDefault(now: now)
        arrivals.append(contentsOf: imported)
        for _ in 0..<3 { arrivals.append(make(at: now)) }
    }

    func advance(minutes: Double) {
        let steps = max(1, Int(minutes.rounded(.up))), dm = minutes / Double(steps)
        for _ in 0..<steps {
            now += dm
            if !DemoState.on, rng.chance((config.capacity["encountersPerHour"] ?? 14) / 60 * dm) { arrivals.append(make(at: now)) }
            capacity.step(now: now, dm: dm)
        }
    }

    func nextItem() -> DomainItem? {
        guard !arrivals.isEmpty else { return nil }
        if DemoState.on && !Set(enc.keys).isSubset(of: completed) { return nil }   // demo: one encounter at a time
        var e = arrivals.removeFirst(); e.arrival = now; enc[e.id] = e
        requestBN(e.id)
        return item(e)
    }

    private func item(_ e: DREncounter) -> DomainItem {
        var f: [String: String] = ["age": "\(e.age)", "sex": e.sex, "cc": e.cc.rawValue, "channel": e.channel,
                                   "companion": e.companion ?? "none", "interpreter": e.interpreter ? "yes" : "no",
                                   "language": e.language, "healthLiteracy": e.literacyLow ? "low" : "adequate",
                                   "aiPlan": e.aiPlan, "vitals": e.vitalsLine]
        for (k, v) in e.fields { f["stated_" + k] = v.stated }
        f["showCapacity"] = e.cc.isOutpatient ? "no" : "yes"
        f["pmh"] = e.pmh.isEmpty ? "none" : e.pmh.joined(separator: ", ")
        let ctx = [e.channel, e.companion.map { "with \($0)" }, e.interpreter ? "interpreter (\(e.language))" : nil].compactMap { $0 }.joined(separator: " · ")
        return DomainItem(id: e.id, fields: f,
                          displayLine: "[SYNTHETIC] \(e.age)\(e.sex) · \(e.cc.rawValue) · \(ctx) · \(e.vitalsLine)",
                          shortLine: "\(e.age)\(e.sex) \(e.cc.rawValue)\(e.trigger.map { " · ⚠ \($0)" } ?? "")",
                          historyLine: (e.pmh.isEmpty ? "none" : e.pmh.joined(separator: ", ")) + " · meds: \(e.fields["meds"]?.stated ?? "—")",
                          arrivalTime: e.arrival)
    }

    // MARK: generator
    private func make(at t: Double) -> DREncounter {
        n += 1
        let edMix = ProcessInfo.processInfo.arguments.contains("-edMix")
        let cc = edMix || !rng.chance(0.75) ? rng.pick(DRComplaint.ed + [.refill]) : rng.pick(OPSpec.outpatient)
        if cc.isOutpatient { return makeOutpatient(cc, at: t) }
        let sex = cc == .uti ? (rng.chance(0.85) ? "F" : "M") : (rng.chance(0.5) ? "F" : "M")
        var age = Int(rng.range(16, 88)); if cc == .headInjury { age = max(age, Int(rng.range(50, 90))) }
        let companion: String? = rng.chance(age >= 75 || age < 18 ? 0.7 : 0.3) ? rng.pick(age < 18 ? ["parent"] : ["partner", "adult child", "friend"]) : nil
        let interp = rng.chance(0.18)
        var e = DREncounter(id: String(format: "E%03d", n), age: age, sex: sex, cc: cc,
                            channel: rng.pick(["text chat", "voice call", "app video", "kiosk"]), companion: companion,
                            interpreter: interp, language: interp ? rng.pick(["Spanish", "Haitian Creole", "Vietnamese", "Arabic"]) : "English",
                            literacyLow: rng.chance(0.2), pmh: [], aiPlan: rng.pick(DRSpec.plans[cc]!), aiDdx: DRSpec.ddx[cc]!, arrival: t)
        // vitals (home device / kiosk) sometimes available
        if rng.chance(0.65) {
            e.hr = Int(rng.range(64, 108)); e.sbp = Int(rng.range(104, 158)); e.spo2 = Int(rng.range(94, 100)); e.temp = rng.range(36.4, 37.8)
            if cc == .cough { e.temp = rng.range(37.6, 39.4); e.hr = Int(rng.range(84, 118)) }
        }
        if rng.chance(0.2) { e.pmh.append("hypertension") }; if rng.chance(0.12) { e.pmh.append("diabetes") }
        if age > 60 && rng.chance(0.2) { e.pmh.append("CAD") }
        // hidden truths
        func addField(_ k: String, benign: String, risky: String, p: Double) {
            e.fields[k] = DRField(name: k, benign: benign, truth: rng.chance(p) ? risky : benign, stated: "—", trueState: .reliable,
                                  asked: false, sensitive: DRSpec.sensitive.contains(k))
        }
        if sex == "F" && age >= 14 && age <= 50 { addField("pregnancy", benign: "not pregnant", risky: "pregnant", p: cc == .abdPain || cc == .uti ? 0.25 : 0.1) }
        let methadoneP = 0.1, anticoagP = age > 65 ? 0.35 : 0.08
        let medTruth = rng.chance(anticoagP) ? "apixaban" : (rng.chance(methadoneP) ? "methadone" : "none")
        e.fields["meds"] = DRField(name: "meds", benign: "none", truth: medTruth, stated: "—", trueState: .reliable, asked: false, sensitive: medTruth == "methadone")
        addField("substance", benign: "none", risky: cc == .chestPain || cc == .palpitations ? "cocaine" : "heavy alcohol", p: 0.18)
        addField("allergies", benign: "NKDA", risky: "penicillin (anaphylaxis)", p: 0.12)
        addField("symptoms", benign: "mild", risky: cc == .chestPain ? "exertional pressure + diaphoresis" : cc == .headache ? "thunderclap, worst ever" : "severe", p: 0.2)
        if [.headInjury, .palpitations].contains(cc) || companion == "partner" { addField("safety", benign: "safe", risky: "unsafe (partner violence)", p: 0.15) }
        if cc == .refill { addField("adherence", benign: "adherent", risky: "missed most doses", p: 0.35) }
        if cc == .uti || cc == .abdPain { addField("sexual", benign: "no new partners", risky: "new partner, unprotected", p: 0.25) }
        if anticoagP > 0.2 && medTruth == "apixaban" { e.pmh.append(rng.chance(0.5) ? "atrial fibrillation" : "") ; e.pmh.removeAll { $0.isEmpty } }
        // conversation: ask / omit / recall error / deliberate nondisclosure, by context
        let exp = Set(DRSpec.expectedFor(cc))
        for k in Array(e.fields.keys) {
            var fd = e.fields[k]!
            fd.asked = rng.chance(exp.contains(k) ? 0.7 : 0.25)
            if !fd.asked { fd.stated = "—"; fd.trueState = .notSaid }
            else {
                var pND = 0.0
                if fd.sensitive && fd.truth != fd.benign {
                    pND = 0.3 + (companion != nil && ["safety", "substance", "sexual", "pregnancy"].contains(k) ? 0.3 : 0)
                        + (interp ? 0.1 : 0) + (age < 18 && ["pregnancy", "sexual", "substance"].contains(k) ? 0.25 : 0) + (e.literacyLow ? 0.05 : 0)
                    if k == "symptoms" { pND = 0.35 + (age > 70 ? 0.15 : 0) }   // minimization
                }
                let pRecall = 0.06 + (age >= 75 && companion == nil ? 0.18 : 0) + (e.literacyLow && ["meds", "allergies"].contains(k) ? 0.12 : 0)
                    + (interp ? 0.08 : 0)
                if rng.chance(min(pND, 0.9)) { fd.stated = fd.benign; fd.trueState = .saidDifferently }
                else if fd.truth != fd.benign && rng.chance(pRecall) { fd.stated = fd.benign; fd.trueState = .saidWrong }
                else if fd.truth == fd.benign && rng.chance(pRecall * 0.3) { fd.stated = "unsure"; fd.trueState = .saidWrong }
                else { fd.stated = fd.truth; fd.trueState = .reliable }
            }
            // consistency clues the system can see
            if fd.truth != fd.benign && fd.stated != fd.truth {
                switch k {
                case "meds" where fd.truth == "methadone": if rng.chance(0.5) { fd.clue = "pharmacy record: methadone clinic dispensing" }
                case "meds" where fd.truth == "apixaban": if e.pmh.contains("atrial fibrillation") { fd.clue = "PMH lists AFib but no anticoagulant stated" }
                case "pregnancy": if rng.chance(0.35) { fd.clue = "transcript: \"my period is a bit late\"" }
                case "safety": if companion == "partner" && rng.chance(0.5) { fd.clue = "partner answered most questions for patient" }
                case "adherence": if rng.chance(0.6) { fd.clue = "refill gap: last fill 70 days ago for 30-day supply" }
                case "substance": if rng.chance(0.3) { fd.clue = "transcript: \"just a few drinks on weekends\" then hesitation" }
                case "symptoms": if rng.chance(0.3) { fd.clue = "transcript: \"it's probably nothing\" + needed to stop walking" }
                default: break
                }
            }
            e.fields[k] = fd
        }
        // hard escalation triggers (unambiguous stated data) — forced on encounter 3 with -forceTrigger
        let force = ProcessInfo.processInfo.arguments.contains("-forceTrigger") && n % 7 == 3
        if force || rng.chance(0.08) {
            switch force ? (n % 2 == 1 ? 0 : 1) : Int(rng.next() % 4) {
            case 0: e.cc = .chestPain; e.trigger = "STEMI: home ECG patch ST elevation + crushing pain"; e.aiDdx = DRSpec.ddx[.chestPain]!
            case 1: e.trigger = "stroke: facial droop + arm weakness, onset 40 min"
            case 2: e.sbp = 78; e.hr = 132; e.spo2 = 89; e.temp = e.temp ?? 37.0; e.trigger = "unstable vitals: SBP 78, HR 132"
            default: e.trigger = "AMS: companion reports new confusion since morning"
            }
        }
        return e
    }

    /// Outpatient AI-doc request: generic fields that apply + complaint-specific hidden truths, same ask/omit/conceal model.
    private func makeOutpatient(_ cc: DRComplaint, at t: Double) -> DREncounter {
        let sex: String = cc == .hrt ? "F" : (cc == .testosterone || cc == .erectile) ? "M" : (rng.chance(0.55) ? "F" : "M")
        let age: Int = cc == .hrt ? Int(rng.range(45, 64)) : cc == .erectile ? Int(rng.range(38, 78)) : cc == .testosterone ? Int(rng.range(25, 66)) : Int(rng.range(19, 72))
        let companion: String? = rng.chance(age >= 75 ? 0.6 : 0.2) ? rng.pick(["partner", "adult child", "friend"]) : nil
        let interp = rng.chance(0.12)
        var e = DREncounter(id: String(format: "E%03d", n), age: age, sex: sex, cc: cc,
                            channel: rng.pick(["text chat", "voice call", "app video"]), companion: companion,
                            interpreter: interp, language: interp ? rng.pick(["Spanish", "Haitian Creole", "Vietnamese", "Arabic"]) : "English",
                            literacyLow: rng.chance(0.15), pmh: [], aiPlan: rng.pick(OPSpec.plans[cc]!), aiDdx: OPSpec.ddx[cc]!, arrival: t)
        if rng.chance(0.4) { e.hr = Int(rng.range(62, 96)); e.sbp = Int(rng.range(108, 150)); e.spo2 = Int(rng.range(96, 100)); e.temp = rng.range(36.4, 37.6) }
        if rng.chance(0.25) { e.pmh.append("hypertension") }; if rng.chance(0.12) { e.pmh.append("diabetes") }
        if cc == .uriAbx {
            let c = Int(rng.range(0, 4.99)), d = Int(rng.range(2, 12))
            e.centor = "Centor/McIsaac \(c) · \(d) days\(d >= 10 ? " (≥10 d)" : "")\(rng.chance(0.15) ? " · double-sickening" : " · no double-sickening")"
        }
        let gen = OPSpec.generic[cc] ?? []
        if gen.contains("meds") {
            let m = rng.chance(0.1) ? "methadone" : "none"
            e.fields["meds"] = DRField(name: "meds", benign: "none", truth: m, stated: "—", trueState: .reliable, asked: false, sensitive: m == "methadone")
        }
        if gen.contains("pregnancy") && sex == "F" && age <= 50 {
            e.fields["pregnancy"] = DRField(name: "pregnancy", benign: "not pregnant", truth: rng.chance(0.08) ? "pregnant" : "not pregnant", stated: "—",
                                            trueState: .reliable, asked: false, sensitive: true)
        }
        if gen.contains("allergies") {
            e.fields["allergies"] = DRField(name: "allergies", benign: "NKDA", truth: rng.chance(0.1) ? "penicillin (anaphylaxis)" : "NKDA", stated: "—",
                                            trueState: .reliable, asked: false, sensitive: false)
        }
        for f in OPSpec.fields[cc] ?? [] {
            e.fields[f.key] = DRField(name: f.key, benign: f.benign, truth: rng.chance(min(0.6, f.p * 2.5)) ? f.risky : f.benign, stated: "—",
                                      trueState: .reliable, asked: false, sensitive: f.sensitive)
        }
        for k in Array(e.fields.keys) {
            var fd = e.fields[k]!
            fd.asked = rng.chance(0.7)
            let risky = fd.truth != fd.benign
            if !fd.asked { fd.trueState = .notSaid }
            else {
                let pND = fd.sensitive && risky ? 0.35 + (companion != nil ? 0.2 : 0) + (interp ? 0.1 : 0) : 0
                let pRecall = 0.06 + (age >= 75 && companion == nil ? 0.15 : 0) + (e.literacyLow ? 0.1 : 0) + (interp ? 0.08 : 0)
                if rng.chance(pND) { fd.stated = fd.benign; fd.trueState = .saidDifferently }
                else if risky && rng.chance(pRecall + (fd.sensitive ? 0 : 0.15)) { fd.stated = fd.benign; fd.trueState = .saidWrong }
                else { fd.stated = fd.truth; fd.trueState = .reliable }
            }
            if risky && fd.stated != fd.truth {
                if let op = OPSpec.field(k), let c = op.clue, rng.chance(0.5) { fd.clue = c }
                if k == "meds", rng.chance(0.5) { fd.clue = "pharmacy record: methadone clinic dispensing" }
            }
            e.fields[k] = fd
        }
        if let tr = OPSpec.triggers[cc], rng.chance(0.05) { e.trigger = rng.pick(tr) }
        return e
    }

    // MARK: Bayesian sidecar
    func requestBN(_ id: String) {
        guard let e = enc[id], let body = BNMap.request(e) else { return }
        Task { @MainActor in
            if let r = await RiskClient.assess(body) { self.bn[id] = r; self.sidecarSeen = true; self.dirty.append(id) }
            else { self.bn[id] = nil }
        }
    }

    func riskBadge(item: DomainItem) -> String? {
        guard let e = enc[item.id] else { return nil }
        if let r = bn[e.id] { return "BN · params: \(r.param_source)" }
        if BNMap.complaint[e.cc] == nil { return "fallback · no BN for complaint" }
        return sidecarOnline == false && !sidecarSeen ? "fallback · sidecar offline" : "fallback · sidecar pending"
    }

    /// BN-backed reliability where the sidecar answered; Swift heuristics for the remaining fields.
    func reliabilityList(_ e: DREncounter) -> [FieldReliability] {
        var base = fallbackReliability(e)
        guard e.trigger == nil, let r = bn[e.id] else { return base }
        for t in r.truths {
            guard let fname = BNMap.field[t.truth], let k = base.firstIndex(where: { $0.field == fname }), let fd = e.fields[fname] else { continue }
            var f = base[k]
            f.state = ReliabilityState(rawValue: t.state) ?? f.state
            f.score = t.report == "pos" || t.answer == "pos" ? 0.95 : max(0.02, 1 - t.posterior)
            f.why = ["BN: P(\(t.truth)) \(t.label) (\(Int((t.posterior * 100).rounded()))%)", t.why, "params: \(r.param_source)"] + f.why.filter { $0.hasPrefix("inconsistent") }
            if let p = fd.physician { f.state = p; f.why.insert("physician re-rated", at: 0) }
            if let w = t.if_wrong, f.state != .reliable || t.posterior >= 0.15 {
                f.ifWrong = RiskRange(alternative: t.truth == "substance" ? "substance use" : t.truth.replacingOccurrences(of: "_", with: " "),
                                      pAlt: t.posterior, riskLo: w.low, riskHi: w.high, severity: w.severity, worstMiss: w.worst_miss,
                                      planChange: w.plan_change, ddxShift: "\(w.dx): now \(w.label_current) · if wrong up to \(w.label_high)")
            } else { f.ifWrong = nil }
            base[k] = f
        }
        return base
    }

    // MARK: reliability estimate (system view: context + consistency, never the hidden truth)
    func fallbackReliability(_ e: DREncounter) -> [FieldReliability] {
        let key = Set(DRSpec.expectedFor(e.cc))
        return e.fields.values.sorted { ($0.name) < ($1.name) }.map { fd in
            var why: [String] = []
            var score = 0.9
            if fd.sensitive { score -= 0.15; why.append("sensitive topic") }
            if let c = e.companion, ["safety", "substance", "sexual", "pregnancy"].contains(fd.name) { score -= 0.2; why.append("\(c) present") }
            if e.interpreter { score -= 0.1; why.append("interpreter (\(e.language))") }
            if e.age < 18 && ["pregnancy", "sexual", "substance"].contains(fd.name) { score -= 0.15; why.append("adolescent") }
            if e.age >= 75 && e.companion == nil && ["meds", "allergies"].contains(fd.name) { score -= 0.15; why.append("age ≥75, no companion (recall)") }
            if e.literacyLow && ["meds", "allergies", "adherence"].contains(fd.name) { score -= 0.1; why.append("low health literacy") }
            if fd.name == "symptoms" && e.age > 70 { score -= 0.1; why.append("older adults minimize") }
            var state: ReliabilityState = .reliable
            var notAsked: [String] = []
            if fd.stated == "—" { state = .notSaid; score = 0.2; notAsked = [DRSpec.q(fd.name)]; why.insert("never asked", at: 0) }
            else if fd.stated == "unsure" { state = .saidWrong; score = 0.35; why.insert("uncertain answer", at: 0) }
            if let c = fd.clue { state = fd.sensitive ? .saidDifferently : .saidWrong; score = 0.25; why.insert("inconsistent: \(c)", at: 0) }
            if fd.confirmed { state = .reliable; score = max(score, 0.9); why.insert(e.asks.contains(fd.name) ? "confirmed by direct question" : "verified in record", at: 0) }
            if let p = fd.physician { state = p; score = p == .reliable ? max(score, 0.85) : min(score, 0.3); why.insert("physician re-rated", at: 0) }
            score = max(0.05, min(0.98, score))
            var r = FieldReliability(field: fd.name, stated: fd.stated, state: state, score: score, why: why, key: key.contains(fd.name),
                                     ifWrong: nil, notAsked: notAsked, physicianState: fd.physician, confirmedByAsk: fd.confirmed,
                                     trueValue: fd.truth, trueState: fd.trueState)
            if score < 0.75 { r.ifWrong = ifWrong(e, fd, score) }
            return r
        }
    }

    /// Alternative-if-wrong: the dangerous value, risk delta range, worst miss, plan change.
    private func ifWrong(_ e: DREncounter, _ fd: DRField, _ score: Double) -> RiskRange? {
        let base: [String: Double] = ["pregnancy": 0.15, "meds": 0.15, "substance": 0.2, "safety": 0.15, "adherence": 0.35,
                                      "allergies": 0.1, "symptoms": 0.25, "sexual": 0.25]
        let pAlt = min(0.9, (1 - score) * (base[fd.name] ?? 0.1) * 2.4)
        var alt = fd.benign == fd.stated || fd.stated == "—" || fd.stated == "unsure" ? "" : fd.benign
        var sev = 2, worst = "", plan = "", ddx = ""
        let p = e.aiPlan
        switch fd.name {
        case "pregnancy":
            alt = "pregnant"; sev = e.cc == .abdPain ? 5 : 3
            worst = e.cc == .abdPain ? "ruptured ectopic" : "teratogenic exposure"
            plan = "hCG before any disposition" + ((p.contains("cipro") || p.contains("ibuprofen")) ? "; replace \(p.components(separatedBy: " ").first!)" : "")
            ddx = e.cc == .abdPain ? "ectopic enters ddx" : "pregnancy-safe meds only"
        case "meds":
            if e.cc == .headInjury { alt = "on anticoagulant"; sev = 5; worst = "delayed ICH"; plan = "CT head now"; ddx = "ICH ↑×4" }
            else if p.contains("azithro") || p.contains("levo") || p.contains("cipro") || p.contains("ondansetron") {
                alt = "on methadone"; sev = 4; worst = "QT prolongation / torsades"; plan = "ECG for QTc; avoid \(p.components(separatedBy: " ").first!)"; ddx = "—" }
            else { alt = "on undisclosed meds"; sev = 2; worst = "interaction"; plan = "med reconciliation"; ddx = "—" }
        case "substance":
            if e.cc == .chestPain || e.cc == .palpitations { alt = "cocaine use"; sev = 4; worst = "cocaine-associated ACS"; plan = "ED eval; no β-blocker"; ddx = "ACS ↑, vasospasm" }
            else { alt = "heavy alcohol"; sev = 3; worst = "withdrawal / pancreatitis"; plan = "CIWA, lipase"; ddx = "pancreatitis enters" }
        case "safety": alt = "unsafe at home"; sev = 4; worst = "return with injury"; plan = "private screen, SW, safe disposition"; ddx = "NAT/IPV injury pattern"
        case "adherence": alt = "non-adherent"; sev = 3; worst = "decompensation (DKA/seizure)"; plan = "labs before refill, adherence plan"; ddx = "decompensation ↑"
        case "allergies":
            alt = "penicillin anaphylaxis"; let beta = p.contains("amox") || p.contains("cephal")
            sev = beta ? 5 : 1; worst = beta ? "anaphylaxis" : "none for current plan"; plan = beta ? "switch antibiotic class" : "none"; ddx = "—"
        case "symptoms":
            alt = "worse than stated"; sev = e.cc == .chestPain || e.cc == .headache ? 5 : 3
            worst = e.cc == .chestPain ? "missed ACS" : e.cc == .headache ? "missed SAH" : "under-triage"
            plan = e.cc == .chestPain ? "ECG + troponin today" : e.cc == .headache ? "ED for CT" : "same-day visit"
            ddx = e.cc == .chestPain ? "ACS 15%→up to 40%" : e.cc == .headache ? "SAH 5%→up to 20%" : "shift to serious causes"
        case "sexual": alt = "STI exposure"; sev = 2; worst = "untreated PID/STI"; plan = "add STI testing"; ddx = "STI/PID ↑"
        default:
            guard let op = OPSpec.field(fd.name) else { return nil }
            alt = op.risky; sev = ["escalate": 5, "decline": 4, "labs first": 3].first { $0.key == op.action }?.value ?? 2
            if op.key == "nitrate_use" { sev = 5 }
            worst = op.worst; plan = op.action; ddx = "—"
        }
        let w = Double(sev) / 5
        return RiskRange(alternative: alt, pAlt: pAlt, riskLo: pAlt * w * 0.6, riskHi: min(1, pAlt * w * 1.4), severity: sev,
                         worstMiss: worst, planChange: plan, ddxShift: ddx)
    }

    /// differential with uncertainty: low-reliability fields widen/shift probabilities
    func ddxWithUncertainty(_ e: DREncounter) -> [(String, Double, Double)] {
        var d = e.aiDdx.map { ($0.0, $0.1, $0.1) }
        for r in reliabilityList(e) {
            guard let w = r.ifWrong else { continue }
            let add: String? = r.field == "pregnancy" && e.cc == .abdPain ? "ectopic" :
                r.field == "meds" && e.cc == .headInjury ? "ICH" : r.field == "substance" && e.cc == .chestPain ? "ACS" :
                r.field == "symptoms" && e.cc == .chestPain ? "ACS" : r.field == "symptoms" && e.cc == .headache ? "SAH" :
                r.field == "substance" && e.cc == .abdPain ? "pancreatitis" : r.field == "sexual" ? "STI" : nil
            guard let a = add else { continue }
            let hi = w.pAlt * 0.8
            if let i = d.firstIndex(where: { $0.0 == a }) { d[i].2 = min(0.9, d[i].2 + hi) } else { d.append((a, 0, hi)) }
            for i in d.indices where d[i].0 != a { d[i].1 = d[i].1 * (1 - hi) }
        }
        return d.sorted { $0.2 > $1.2 }
    }

    // MARK: ground truth / proposals
    func trueESI(_ e: DREncounter) -> Int {
        if e.trigger != nil { return e.trigger!.hasPrefix("unstable") ? 1 : 2 }
        if e.truthIs("pregnancy", "pregnant") && e.cc == .abdPain { return 2 }
        if e.truthIs("meds", "apixaban") && e.cc == .headInjury { return 2 }
        if e.truthRisky("symptoms") && [.chestPain, .headache].contains(e.cc) { return 2 }
        if e.truthIs("substance", "cocaine") && e.cc == .chestPain { return 2 }
        return statedESI(e)
    }
    func statedESI(_ e: DREncounter) -> Int {
        if let s = e.sbp, s < 90 { return 1 }
        switch e.cc {
        case .chestPain: return e.age >= 40 ? 2 : 3
        case .headInjury: return e.age >= 65 ? 3 : 4
        case .abdPain, .headache, .palpitations: return 3
        case .cough: return (e.spo2 ?? 97) < 92 ? 2 : 4
        case .uti, .refill: return 5
        case .uriAbx: return (e.spo2 ?? 98) < 92 ? 2 : 5
        case .glp1, .hrt, .testosterone, .erectile: return 5
        }
    }
    func truePlan(_ e: DREncounter) -> String {
        if e.trigger != nil { return "escalate" }
        if e.cc.isOutpatient { return outpatientPlan(e) }
        let dxDanger = (e.truthIs("pregnancy", "pregnant") && e.cc == .abdPain) || (e.truthIs("meds", "apixaban") && e.cc == .headInjury)
            || (e.truthRisky("symptoms") && [.chestPain, .headache].contains(e.cc)) || (e.truthIs("substance", "cocaine") && e.cc == .chestPain)
            || e.truthRisky("safety")
        if dxDanger { return "escalate" }
        let p = e.aiPlan
        let planHazard = (e.truthIs("meds", "methadone") && (p.contains("azithro") || p.contains("levo") || p.contains("cipro") || p.contains("ondansetron")))
            || (e.truthIs("pregnancy", "pregnant") && (p.contains("cipro") || p.contains("ibuprofen")))
            || (e.truthRisky("allergies") && (p.contains("amox") || p.contains("cephal"))) || e.truthRisky("adherence")
            || (e.truthIs("substance", "cocaine") && p.contains("propranolol"))
        return planHazard ? "with conditions" : "sign off"
    }
    func outpatientPlan(_ e: DREncounter) -> String {
        var acts: [String] = []
        for (k, fd) in e.fields where fd.truth != fd.benign {
            if let op = OPSpec.field(k) { acts.append(op.action); continue }
            let p = e.aiPlan
            switch k {
            case "meds" where ["azithro", "levo"].contains { p.contains($0) }: acts.append("with conditions")
            case "pregnancy": acts.append(e.cc == .glp1 ? "decline" : (p.contains("levo") ? "with conditions" : "sign off"))
            case "allergies" where p.contains("amox"): acts.append("with conditions")
            default: break
            }
        }
        return acts.max { (OPSpec.actionRank[$0] ?? 0) < (OPSpec.actionRank[$1] ?? 0) } ?? "sign off"
    }

    func groundTruth(item: DomainItem, link: LinkSpec, routed: [String: String]) -> String? {
        guard let e = enc[item.id] else { return nil }
        switch link.id { case "esi": return "\(trueESI(e))"; case "plan": return truePlan(e); default: return nil }
    }
    func propose(item: DomainItem, link: LinkSpec, routed: [String: String]) -> Proposal {
        guard let e = enc[item.id] else { return Proposal(answer: link.options[0], confidence: 0.5) }
        switch link.id {
        case "esi":
            var x = statedESI(e); if rng.chance(0.1) { x = min(5, max(1, x + (rng.chance(0.5) ? 1 : -1))) }
            return Proposal(answer: "\(x)", confidence: rng.range(0.62, 0.95))
        case "plan":
            if DemoState.on { let k = min(max((Int(e.id.dropFirst()) ?? 1) - 1, 0), 5)
                return Proposal(answer: "sign off", confidence: [0.82, 0.86, 0.84, 0.8, 1.0, 0.95][k]) }
            return Proposal(answer: rng.chance(0.08) ? "with conditions" : "sign off", confidence: rng.range(0.6, 0.93))
        default: return Proposal(answer: link.options[0], confidence: 0.5)
        }
    }

                           isolation: false, arrival: now, now: now)
        }
    }

    func hiddenTruth(item: DomainItem) -> [String: String] {
        guard let e = enc[item.id] else { return [:] }

    func askNext(item: DomainItem) -> [VOIQuestion] {
        guard let e = enc[item.id], e.trigger == nil else { return [] }
        if let r = bn[e.id], r.voi.contains(where: { $0.voi > 0.0005 && BNMap.field[$0.truth].map { !e.asks.contains($0) } == true }) {
            return r.voi.compactMap { q -> VOIQuestion? in
                guard let f = BNMap.field[q.truth], !e.asks.contains(f), q.voi > 0.0005 else { return nil }
                return VOIQuestion(id: "\(e.id)-\(f)", field: f, text: q.question, voi: q.voi,
                                   rationale: "BN: expected-loss reduction \(String(format: "%.2f", q.voi)) · P(answer rules out critical) \(Int(q.p_ruleout * 100))%")
            }.prefix(3).map { $0 }
        }
        return reliabilityList(e).compactMap { r -> VOIQuestion? in
            guard let w = r.ifWrong, !r.confirmedByAsk, !e.asks.contains(r.field) else { return nil }
            let planChange = w.planChange == "none" || w.planChange == "sign off" ? 0.1 : 1.0
            let voi = w.pAlt * planChange * Double(w.severity)
            return VOIQuestion(id: "\(e.id)-\(r.field)", field: r.field, text: DRSpec.q(r.field), voi: voi,
                               rationale: "P(wrong) \(Int(w.pAlt * 100))% × plan change \(w.planChange) × severity \(w.severity)")
        }.sorted { $0.voi > $1.voi }.prefix(3).map { $0 }
    }
    func ask(item: DomainItem, questionId: String) -> String? {
        guard var e = enc[item.id], let field = questionId.split(separator: "-").last.map(String.init), var fd = e.fields[field] else { return nil }
        asksSent += 1; e.asks.append(field)
        // simulated patient answers per hidden truth; direct private physician questions reveal more, but not always
        let reveal = DemoState.on || fd.truth == fd.benign || rng.chance(fd.sensitive ? 0.7 : 0.9)
        fd.stated = reveal ? fd.truth : fd.benign
        fd.asked = true; fd.confirmed = reveal; fd.clue = nil
        if reveal && fd.truth != fd.benign { revealed += 1 }
        e.fields[field] = fd; enc[e.id] = e
        requestBN(e.id)
        return reveal ? "Pt: \(fd.truth)" : "Pt: \(fd.benign) (hesitant)"
    }

        e.fields[field]?.physician = state; enc[e.id] = e; rerates += 1; dirty.append(e.id)
    }

    func transcript(item: DomainItem) -> [String] {
        guard let e = enc[item.id] else { return [] }
        var t = ["AI: What brings you in today?", "Pt: \(e.cc.rawValue)\(e.trigger.map { " — \($0)" } ?? "")."]
        let order = ["symptoms", "meds", "pregnancy", "substance", "allergies", "safety", "adherence", "sexual"] + (OPSpec.fields[e.cc] ?? []).map(\.key)
        for k in order {
            guard let fd = e.fields[k], fd.asked else { continue }
            t.append("AI: \(DRSpec.q(k).components(separatedBy: "?").first!)?")
            t.append("Pt: " + (fd.stated == "unsure" ? "I'm not sure…" : fd.stated) + (fd.clue.flatMap { $0.hasPrefix("transcript") ? " " + $0.replacingOccurrences(of: "transcript: ", with: "") : nil } ?? ""))
        }
        for a in e.asks { if let fd = e.fields[a] { t.append("MD: \(DRSpec.q(a))"); t.append("Pt: \(fd.stated)") } }
        return t
    }

    func reasons(item: DomainItem, link: LinkSpec, proposal: String, upstream: [String: String]) -> [ReasonChip] {
        guard let e = enc[item.id] else { return [] }
        if let t = e.trigger { return [ReasonChip(text: "HARD TRIGGER: \(t)", key: "hard trigger", severity: .high)] }
        var r: [ReasonChip] = []
        if let n = e.fields["nitrate_use"], n.stated != n.benign && n.stated != "—" {
            r.append(ReasonChip(text: "HARD STOP: PDE5 inhibitor + nitrate (\(n.stated)) — do not prescribe", key: "HARD STOP", severity: .high))
        }
        let rel = reliabilityList(e)
        for f in rel where f.key && f.state != .reliable {
            r.append(ReasonChip(text: "\(f.field): \(f.state.rawValue)", key: f.state.rawValue,
                                severity: (f.ifWrong?.severity ?? 0) >= 4 ? .high : .warn))
        }
        if let b = bn[e.id] {
            if b.escalation.recommend {
                r.append(ReasonChip(text: "BN: P(critical miss) \(b.p_critical_label) → recommend escalation", key: "recommend escalation", severity: .high))
            }
        } else if rel.contains(where: { ($0.ifWrong?.severity ?? 0) >= 4 && ($0.ifWrong?.pAlt ?? 0) >= 0.2 && !$0.confirmedByAsk && $0.physicianState != .reliable }) {
            r.append(ReasonChip(text: "critical miss not excluded → recommend escalation", key: "recommend escalation", severity: .high))
        }
        if let s = e.sbp, s < 95 { r.append(ReasonChip(text: "SBP \(s)", key: "abnormal vitals", severity: .high)) }
        return r
    }
    func urgency(item: DomainItem, link: LinkSpec, proposal: String) -> Double {
        guard let e = enc[item.id] else { return 0 }
        let w = reliabilityList(e).compactMap { $0.ifWrong }.map { $0.riskHi }.max() ?? 0
        return w * 1.5 + (link.id == "plan" ? 0.2 : 0)
    }
    func downstreamPreview(item: DomainItem, link: LinkSpec, answer: String) -> String? {
        guard let e = enc[item.id] else { return nil }
        switch (link.id, answer) {
        case ("plan", "sign off"): return "Orders: \(e.cc.isOutpatient ? "prescription sent" : "AI plan released to patient") — \(e.aiPlan)"
        case ("plan", "with conditions"): return "Orders: prescribe with changes (\(reliabilityList(e).compactMap { $0.ifWrong?.planChange }.first ?? "edit plan"))"
        case ("plan", "labs first"): return "Orders: request held; labs / measurement / evaluation ordered first"
        case ("plan", "decline"): return "Orders: request declined — patient message with reason + alternative/referral"
        case ("plan", "escalate"):
            return e.trigger != nil ? "Alert Team already paged; ED \((capacity.rooms["resus"] ?? 0) - capacity.used("resus")) resus free" :
                "Alert Team: physician callback ≤15 min; ED main \(capacity.used("main ED"))/\(capacity.rooms["main ED"] ?? 0)"
        case ("esi", let x): return "ED Intake: ESI \(x) → \(Int(x) ?? 3 <= 2 ? "send to ED now" : Int(x) ?? 3 == 3 ? "same-day visit" : "virtual follow-up")"
        default: return nil
        }
    }
    func detail(item: DomainItem, section: String) -> [(String, String)]? {
        guard let e = enc[item.id] else { return nil }
        switch section {
        case "Transcript": return transcript(item: item).enumerated().map { ("\($0.offset + 1)", $0.element) }
        case "Extracted fields":
            return [("CC", e.cc.rawValue), ("vitals", e.vitalsLine), ("PMH", e.pmh.isEmpty ? "none" : e.pmh.joined(separator: ", "))]
                + e.fields.values.sorted { $0.name < $1.name }.map { ($0.name, $0.stated) }
        case "Context":
            return [("age/sex", "\(e.age)\(e.sex)"), ("channel", e.channel), ("companion", e.companion ?? "none"),
                    ("language", e.interpreter ? "\(e.language) via interpreter" : e.language), ("health literacy", e.literacyLow ? "low" : "adequate")]
        case "AI differential (with uncertainty)":
            return ddxWithUncertainty(e).map { ($0.0, $0.1 == $0.2 ? "\(Int($0.1 * 100))%" : "\(Int($0.1 * 100))–\(Int($0.2 * 100))%") }
        case "AI triage + plan":
            let alt = reliabilityList(e).contains { $0.ifWrong != nil && ($0.ifWrong!.severity >= 4) } ? " (ESI \(max(1, statedESI(e) - 1)) if key field wrong)" : ""
            return [("ESI", "\(statedESI(e))\(alt)"), ("plan", e.aiPlan)]
        case "Scores + counselling":
            let rows = (e.centor.map { [("Centor/McIsaac", $0)] } ?? []) + (OPSpec.counselling[e.cc] ?? [])
            return rows.isEmpty ? nil : rows
        case "Expected questions not asked":
            let miss = reliabilityList(e).filter { $0.state == .notSaid }.flatMap { $0.notAsked }
            return miss.isEmpty ? [("—", "all expected questions asked")] : miss.map { ("☐", $0) }
        default: return nil
        }
    }

    func applicableLinks(item: DomainItem) -> [String]? { enc[item.id].map { $0.cc.isOutpatient ? ["plan"] : nil } ?? nil }
    func demoCaption(item: DomainItem) -> String? { enc[item.id]?.demoCaption }
    var isDemoDone: Bool { demoDone }

         MetricValue(label: "ED census", value: "main \(capacity.used("main ED"))/\(capacity.rooms["main ED"] ?? 0)")]
    }

        var c: [String: Int] = [:]

        return ReliabilityState.allCases.filter { $0 != .reliable }.map { ($0.rawValue, c[$0.rawValue] ?? 0) }
    }

import Foundation

/// Importer hook for real encounter spreadsheets (one row per encounter).
/// xlsx: export the sheet to CSV (UTF-8) and drop it in the app's Documents as `encounters.csv`, or pass `-importCSV <path>`.
/// Native .xlsx parsing is a TODO (needs a zip/XML reader); the column mapping below is the contract.
/// See docs/AI_DOC_REVIEW.md for the full column table.
enum EncounterImporter {
    static let columns = ["encounter_id", "age", "sex", "chief_complaint", "channel", "companion", "interpreter", "language",
                          "health_literacy", "hr", "sbp", "spo2", "temp", "pmh", "ai_plan", "transcript",
                          "stated_pregnancy", "stated_meds", "stated_substance", "stated_allergies", "stated_symptoms",
                          "stated_safety", "stated_adherence", "stated_sexual",
                          "true_pregnancy", "true_meds", "true_substance", "true_allergies", "true_symptoms",
                          "true_safety", "true_adherence", "true_sexual", "hard_trigger"]

    @MainActor static func loadDefault(now: Double) -> [DREncounter] {
        let a = ProcessInfo.processInfo.arguments
        let docs = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0].appendingPathComponent("encounters.csv")
        let url = a.firstIndex(of: "-importCSV").flatMap { $0 + 1 < a.count ? URL(fileURLWithPath: a[$0 + 1]) : nil } ?? docs
        guard let text = try? String(contentsOf: url, encoding: .utf8) else { return [] }
        return parse(csv: text, now: now)
    }

    static func parse(csv: String, now: Double) -> [DREncounter] {
        var lines = csv.split(whereSeparator: \.isNewline).map(String.init)
        guard !lines.isEmpty else { return [] }
        let header = split(lines.removeFirst()).map { $0.lowercased().trimmingCharacters(in: .whitespaces) }
        return lines.compactMap { line in
            let v = split(line); var r: [String: String] = [:]
            for (i, h) in header.enumerated() where i < v.count { r[h] = v[i].trimmingCharacters(in: .whitespaces) }
            guard let id = r["encounter_id"], !id.isEmpty else { return nil }
            let cc = DRComplaint.allCases.first { $0.rawValue.lowercased() == (r["chief_complaint"] ?? "").lowercased() } ?? .chestPain
            var e = DREncounter(id: "X-" + id, age: Int(r["age"] ?? "") ?? 40, sex: r["sex"] ?? "U", cc: cc, channel: r["channel"] ?? "imported",
                                companion: (r["companion"]).flatMap { $0.isEmpty || $0 == "none" ? nil : $0 }, interpreter: r["interpreter"] == "yes",
                                language: r["language"] ?? "English", literacyLow: r["health_literacy"] == "low",
                                hr: Int(r["hr"] ?? ""), sbp: Int(r["sbp"] ?? ""), spo2: Int(r["spo2"] ?? ""), temp: Double(r["temp"] ?? ""),
                                pmh: (r["pmh"] ?? "").split(separator: ";").map { $0.trimmingCharacters(in: .whitespaces) },
                                aiPlan: r["ai_plan"] ?? "—", aiDdx: DRSpec.ddx[cc]!, arrival: now)
            let benign = ["pregnancy": "not pregnant", "meds": "none", "substance": "none", "allergies": "NKDA", "symptoms": "mild",
                          "safety": "safe", "adherence": "adherent", "sexual": "no new partners"]
            for (k, b) in benign {
                guard let s = r["stated_" + k], !s.isEmpty else { continue }
                let t = r["true_" + k].flatMap { $0.isEmpty ? nil : $0 } ?? s   // no truth column → treat stated as truth (unknown)
                let st: ReliabilityState = s == "—" ? .notSaid : (t == s ? .reliable : .saidWrong)
                e.fields[k] = DRField(name: k, benign: b, truth: t, stated: s, trueState: st, asked: s != "—", clue: nil,
                                      sensitive: DRSpec.sensitive.contains(k))
            }
            if let t = r["hard_trigger"], !t.isEmpty { e.trigger = t }
            return e
        }
    }
    /// minimal CSV split with quoted fields
    static func split(_ line: String) -> [String] {
        var out: [String] = [], cur = "", q = false
        for ch in line {
            if ch == "\"" { q.toggle() } else if ch == "," && !q { out.append(cur); cur = "" } else { cur.append(ch) }
        }
        out.append(cur); return out
    }
}

enum RiskClient {
    static var baseURL = URL(string: ProcessInfo.processInfo.environment["RISK_URL"] ?? "http://127.0.0.1:8765")!
    static let session: URLSession = {
        let c = URLSessionConfiguration.ephemeral
        c.timeoutIntervalForRequest = 0.8; c.timeoutIntervalForResource = 1.5
        return URLSession(configuration: c)
    }()
    /// one-shot health ping (demo start): sidecar reachable?
    static func checkHealth(_ done: @escaping @MainActor (Bool) -> Void) {
        Task { @MainActor in
            if ProcessInfo.processInfo.arguments.contains("-noSidecar") { done(false); return }
            let ok = (try? await session.data(from: baseURL.appendingPathComponent("health"))).map { ($0.1 as? HTTPURLResponse)?.statusCode == 200 } ?? false
            done(ok)
        }
    }
    static func assess(_ body: [String: Any]) async -> BNResult? {
        guard !ProcessInfo.processInfo.arguments.contains("-noSidecar"),
              let data = try? JSONSerialization.data(withJSONObject: body) else { return nil }
        var rq = URLRequest(url: baseURL.appendingPathComponent("assess"))
        rq.httpMethod = "POST"; rq.httpBody = data; rq.setValue("application/json", forHTTPHeaderField: "Content-Type")
        guard let (d, resp) = try? await session.data(for: rq), (resp as? HTTPURLResponse)?.statusCode == 200 else { return nil }
        return try? JSONDecoder().decode(BNResult.self, from: d)
    }

enum BNMap {

                                                   .cough: "infection_abx", .uti: "infection_abx",

                                             "chest_pain": ["substance", "minimized", "qt_med"], "head_injury": ["anticoag", "substance", "unsafe_home"],

                                             "uri_abx": ["qt_med", "pregnant", "pcn_allergy", "airway_red_flag"],

                                          "unsafe_home": "safety", "minimized": "symptoms", "pcn_allergy": "allergies"]
        .merging(Dictionary(uniqueKeysWithValues: OPSpec.fields.values.joined().map { ($0.key, $0.key) })) { a, _ in a }
    static let hedgeTruths: Set<String> = ["pregnant", "substance", "unsafe_home", "minimized"]

    @MainActor static func request(_ e: DREncounter) -> [String: Any]? {
        guard let c = complaint[e.cc], let ts = truths[c] else { return nil }
        var reports: [String: Any] = [:], signals: [String: Any] = [:], answers: [String: Any] = [:], known: [String: Any] = [:]
        for t in ts {
            let fname = field[t]!
            guard let fd = e.fields[fname] else {
                if t == "pregnant" { known[t] = "no" }       // not eligible (sex/age) → known not pregnant
                else if t == "unsafe_home" { reports[t] = NSNull() } else { reports[t] = NSNull() }
                continue
            }
            // report
            switch (t, fd.stated) {
            case (_, "—"): reports[t] = NSNull()
            case ("qt_med", let s): reports[t] = s == "methadone" ? "pos" : "neg"
            case ("anticoag", let s): reports[t] = s == "apixaban" ? "pos" : "neg"
            case ("substance", let s): reports[t] = s == "cocaine" ? "cocaine" : s == "heavy alcohol" ? "alcohol" : "neg"
            case (_, let s): reports[t] = (s == fd.benign || s == "unsure") ? "neg" : "yes"     // disclosed → truth fixed
            }
            if let sig = OPSpec.field(t)?.signal { signals[t] = [sig: fd.clue != nil] }   // system checked record/transcript
            if fd.confirmed && !e.asks.contains(fname) && fd.truth == fd.benign { known[t] = "no" }   // verified in record
            else if let clue = fd.clue {
                if hedgeTruths.contains(t) && (clue.hasPrefix("transcript") || clue.contains("answered")) { signals[t] = ["hedge": true] }
                if (t == "qt_med" || t == "anticoag") { signals[t] = ["record": true] }
            }
            if e.asks.contains(fname) {
                let relevant = (t == "qt_med" && fd.truth == "methadone") || (t == "anticoag" && fd.truth == "apixaban") || !["qt_med", "anticoag"].contains(t)
                answers[t] = fd.confirmed && fd.truth != fd.benign && relevant ? "pos" : "neg"
            }
        }
        let p = e.aiPlan
        var flags: [String: Any] = [:]
        let flagNames: [String: [String]] = ["abdominal_pain": ["plan_teratogen"], "chest_pain": ["plan_qt_drug"], "head_injury": [],
                                             "infection_abx": ["plan_qt_drug", "plan_teratogen"], "uri_abx": ["plan_qt_drug", "plan_teratogen"]]
        for f in flagNames[c] ?? [] {
            flags[f] = f == "plan_qt_drug" ? ["azithro", "levo", "cipro", "ondansetron"].contains { p.contains($0) }
                                           : ["cipro", "ibuprofen", "doxy"].contains { p.contains($0) }
        }
        return ["complaint": c, "encounter_id": e.id,
                "context": ["companion": e.companion != nil, "interpreter": e.interpreter, "adolescent": e.age < 18,
                            "low_literacy": e.literacyLow, "elderly_alone": e.age >= 75 && e.companion == nil],
                "plan_flags": flags, "reports": reports, "signals": signals, "answers": answers, "known": known]
    }

import Foundation
/// Discrete-event ED capacity model (SYNTHETIC). Rooms per area, ESI service times,
/// admissions boarding until inpatient beds release, waiting room + LWBS, door-to-room.
struct EDCapacity {
    struct Occ { var id: String; var esi: Int; var doneAt: Double; var admit: Bool; var boarding = false }
    struct Waiter { var id: String; var esi: Int; var area: String; var arrival: Double; var iso: Bool }
    var rooms: [String: Int]
    var occ: [String: [Occ]] = [:]
    var wr: [Waiter] = []
    var lwbs = 0, boardingMinutes = 0.0, alertWaits: [Double] = [], doorToRoom: [Double] = []
    var placed: [String: Int] = [:]
    var bedRate: Double, lwbsAfter: Double, lwbsPerMin: Double, bedCredit = 0.0
    var rng: SplitMix64
    static let areas = ["resus", "main ED", "fast track", "isolation"]
    init(params p: [String: Double], seed: UInt64) {
        rooms = ["resus": Int(p["resusRooms"] ?? 3), "main ED": Int(p["mainRooms"] ?? 12),
                 "fast track": Int(p["fastRooms"] ?? 5), "isolation": Int(p["isoRooms"] ?? 3)]
        bedRate = p["inpatientBedsPerHour"] ?? 2.5; lwbsAfter = p["lwbsAfterMin"] ?? 120; lwbsPerMin = p["lwbsPerMin"] ?? 0.008
        rng = SplitMix64(s: seed &+ 77)
    }
    var waitingCount: Int { wr.count }
    func used(_ a: String) -> Int { occ[a]?.count ?? 0 }
    func free(_ a: String) -> Bool { used(a) < (rooms[a] ?? 0) }
    func waiting(_ a: String) -> Int { wr.filter { $0.area == a }.count }
    var boardingNow: Int { occ.values.flatMap { $0 }.filter { $0.boarding }.count }
    mutating func rnd() -> Double { Double(rng.next() % 1_000_000) / 1_000_000 }
    mutating func service(_ esi: Int) -> Double { [1: 240.0, 2: 200, 3: 170, 4: 80, 5: 50][esi, default: 150] * (0.6 + 0.8 * rnd()) }
    func admitP(_ esi: Int) -> Double { [1: 0.85, 2: 0.6, 3: 0.3, 4: 0.05, 5: 0.01][esi, default: 0.2] }
    static func physical(_ area: String) -> String { areas.contains(area) || area == "waiting room" ? area : "main ED" }
    mutating func prewarm(now: Double) {
        for (a, n) in [("resus", 2), ("main ED", 10), ("fast track", 3), ("isolation", 1)] {
            for i in 0..<n {
                let esi = a == "resus" ? 1 : a == "fast track" ? 4 : 3
                var o = Occ(id: "pre-\(a)-\(i)", esi: esi, doneAt: now + service(esi) * rnd(), admit: rnd() < admitP(esi))
                if a == "main ED" && i < 3 { o.admit = true; o.boarding = true }
                occ[a, default: []].append(o)
            }
        }
    }
    mutating func step(now: Double, dm: Double) {
        bedCredit += bedRate * dm / 60
        for a in Self.areas {
            var keep: [Occ] = []
            for var o in occ[a] ?? [] {
                if o.boarding { boardingMinutes += dm; if bedCredit >= 1 { bedCredit -= 1; continue }; keep.append(o); continue }
                if now >= o.doneAt { if o.admit { o.boarding = true; keep.append(o) }; continue }
                keep.append(o)
            }
            occ[a] = keep
        }
        bedCredit = min(bedCredit, 3)
        // LWBS (low acuity long waits) and rooming from waiting room by ESI then arrival
        var stay: [Waiter] = []
        for w in wr {
            if w.esi >= 3 && now - w.arrival > lwbsAfter && rnd() < lwbsPerMin * dm { lwbs += 1 } else { stay.append(w) }
        }
        wr = stay
        wr.sort { ($0.esi, $0.arrival) < ($1.esi, $1.arrival) }
        var rest: [Waiter] = []
        for w in wr {
            let target = free(w.area) ? w.area : (w.esi <= 2 && free("resus") ? "resus" : nil)
            if let t = target { room(w.id, w.esi, t, w.arrival, now) } else { rest.append(w) }
        }
        wr = rest
    }
    mutating func room(_ id: String, _ esi: Int, _ a: String, _ arrival: Double, _ now: Double) {
        occ[a, default: []].append(Occ(id: id, esi: esi, doneAt: now + service(esi), admit: rnd() < admitP(esi)))
        doorToRoom.append(now - arrival)
    }
    /// Capacity-aware ground truth for placement.
    func capacityAwareArea(esi: Int, alert: String, base: String) -> String {
        let a = Self.physical(base)
        if a == "waiting room" || free(a) { return base }
        if esi <= 2 && free("resus") { return "resus" }
        if esi <= 1 { return "resus" }
        return "waiting room"
    }
    mutating func place(id: String, esi: Int, area: String, isolation: Bool, arrival: Double, now: Double) {
        placed[area, default: 0] += 1
        var a = Self.physical(area)
        if a == "waiting room" { a = isolation ? "isolation" : (esi <= 3 ? "main ED" : "fast track"); wr.append(Waiter(id: id, esi: esi, area: a, arrival: arrival, iso: isolation)); return }
        if free(a) { room(id, esi, a, arrival, now) } else { wr.append(Waiter(id: id, esi: esi, area: a, arrival: arrival, iso: isolation)) }
    }
    func snapshot(queue: Int) -> [String: String] {
        var d = ["triageQueue": "\(queue)", "waitingRoom": "\(wr.count)", "boarding": "\(boardingNow)", "lwbs": "\(lwbs)"]
        for a in Self.areas { d[a] = "\(used(a))/\(rooms[a] ?? 0)"; d[a + " queue"] = "\(waiting(a))" }
        return d
    }
    func line(queue: Int) -> String {
        func f(_ n: String, _ a: String) -> String { let w = waiting(a); return "\(n) \(used(a))/\(rooms[a] ?? 0)" + (w > 0 ? " (+\(w))" : "") }
        return [f("resus", "resus"), f("main", "main ED"), f("FT", "fast track"), f("iso", "isolation"),
                "boarding \(boardingNow)", "WR \(wr.count)", "triage \(queue)"].joined(separator: " · ")
    }
    var medianDoorToRoom: Double { let s = doorToRoom.sorted(); return s.isEmpty ? 0 : s[s.count / 2] }
    var medianAlertWait: Double { let s = alertWaits.sorted(); return s.isEmpty ? 0 : s[s.count / 2] }
    var placedCounts: [(String, Int)] {
        let base = ["resus", "main ED", "fast track", "waiting room"]
        return base.map { ($0, placed[$0] ?? 0) } + placed.keys.filter { !base.contains($0) }.sorted().map { ($0, placed[$0]!) }
    }
}

import Foundation

/// ED triage plug-in (SYNTHETIC). Generator + rule hooks; everything else lives in ed_triage.domain.json.
enum Complaint: String, CaseIterable {
    case chestPain = "chest pain", sob = "SOB", abdPain = "abd pain", fever, headache
    case fallHead = "fall/head strike", neuro = "weakness/neuro deficit", laceration, ankle = "ankle injury", psych, uri = "sore throat/URI"
}

struct EDPatient {
    var id: String; var age: Int; var sex: String; var complaint: Complaint
    var hr: Int, sbp: Int, rr: Int, spo2: Int, temp: Double, gcs: Int
    var ecgSTE = false, onsetHours = 48.0, suicidal = false
    var anticoag = false, immuno = false, chf = false, ckd = false, priorStrokeMI = false
    var mdro = false, bounceback72h = false, recentSurgery = false, baselineAMS = false, pregnant = false
    var arrival: Double = 0

    var vitals: String { "HR \(hr) BP \(sbp)/\(Int(Double(sbp) * 0.62)) RR \(rr) SpO2 \(spo2)% T \(String(format: "%.1f", temp)) GCS \(gcs)" }
    var extra: String {
        switch complaint {
        case .chestPain: return ecgSTE ? " · ECG: STE" : " · ECG: no STE"
        case .neuro: return " · onset \(Int(onsetHours)) h"
        case .psych: return suicidal ? " · SI+" : " · SI−"
        default: return ""
        }
    }
    var historyFlags: [String] {
        var h: [String] = []
        if anticoag { h.append("anticoag") }; if immuno { h.append("immunosuppressed") }
        if chf { h.append("CHF") }; if ckd { h.append("CKD") }; if priorStrokeMI { h.append("prior stroke/MI") }
        if mdro { h.append("MDRO") }; if bounceback72h { h.append("72h bounceback") }
        if recentSurgery { h.append("recent surgery") }; if baselineAMS { h.append("baseline AMS") }
        if pregnant { h.append("pregnant") }
        return h
    }
    var fields: [String: String] {
        var f: [String: String] = ["age": "\(age)", "sex": sex, "complaint": complaint.rawValue, "hr": "\(hr)", "sbp": "\(sbp)",
                                   "rr": "\(rr)", "spo2": "\(spo2)", "temp": String(format: "%.1f", temp), "gcs": "\(gcs)"]
        if complaint == .chestPain { f["ecgSTE"] = ecgSTE ? "yes" : "no" }
        if complaint == .neuro { f["onsetHours"] = "\(Int(onsetHours))" }
        if complaint == .psych { f["suicidal"] = suicidal ? "yes" : "no" }
        for h in historyFlags { f["hx_" + h.replacingOccurrences(of: " ", with: "_")] = "yes" }
        return f
    }
}

enum EDRules {
    static func sirs(_ p: EDPatient) -> Int { (p.temp >= 38.3 || p.temp < 36 ? 1 : 0) + (p.hr > 90 ? 1 : 0) + (p.rr > 20 ? 1 : 0) }
    static let infection: Set<Complaint> = [.fever, .sob, .uri, .abdPain]
    static func sepsis(_ p: EDPatient, hx: Bool = true) -> Bool {
        if hx && p.immuno && p.temp >= 38.0 { return true }
        return infection.contains(p.complaint) && sirs(p) >= 2 && (p.temp >= 38.3 || p.temp < 36)
    }
    static func alert(_ p: EDPatient) -> String {
        if p.complaint == .neuro && p.onsetHours <= 24 { return "stroke" }
        if p.complaint == .chestPain && p.ecgSTE { return "STEMI" }
        return "none"
    }
    static func isolation(_ p: EDPatient, hx: Bool = true) -> Bool {
        (hx && p.mdro) || (p.temp >= 38.0 && [.sob, .uri, .fever].contains(p.complaint))
    }
    static func esi(_ p: EDPatient, hx: Bool = true) -> Int {
        if p.sbp < 80 || p.spo2 < 85 || p.rr > 35 || p.gcs < 9 { return 1 }
        let gcsBaseline = hx && p.baselineAMS && p.gcs >= 13
        let high = alert(p) != "none" || sepsis(p, hx: hx) || p.spo2 < 92 || p.hr > 130
            || (p.complaint == .chestPain && p.age >= 40) || (p.complaint == .psych && p.suicidal)
            || (p.gcs < 13 && !gcsBaseline)
            || (hx && p.pregnant && p.complaint == .abdPain) || (hx && p.anticoag && p.complaint == .fallHead)
            || (hx && p.immuno && p.temp >= 38.0)
        var e: Int
        if high { e = 2 } else {
            switch p.complaint {
            case .laceration, .ankle: e = 4
            case .uri: e = 5
            case .fallHead: e = p.age >= 65 ? 3 : 4
            default: e = 3
            }
        }
        if hx && p.bounceback72h && e >= 3 { e = max(2, e - 1) }
        return e
    }
    static func baseArea(esi: Int, alert: String, isolation: Bool) -> String {
        if esi == 1 || alert != "none" { return "resus" }
        if esi <= 3 { return "main ED" }
        if esi == 4 { return isolation ? "main ED" : "fast track" }
        return isolation ? "fast track" : "waiting room"
    }
}

@MainActor final class EDTriageDomain: DomainModule {
    let config: DomainConfig
    private var rng = SplitMix64(s: 1)
    private var n = 0
    private(set) var now: Double = 0
    var minutesPerSecond: Double { config.capacity["minutesPerSecond"] ?? 1 }
    var patients: [String: EDPatient] = [:]
    private var arrivals: [EDPatient] = []
    var capacity: EDCapacity   // internal so EDReview can read it

    init() {
        config = (try? DomainLoader.config(named: "ed_triage.domain")) ?? DomainConfig(id: "ed_triage", title: "ED (config missing)", agents: [], links: [])
        capacity = EDCapacity(params: config.capacity, seed: 7)
    }

    func reset(seed: UInt64) {
        rng = SplitMix64(s: seed); n = 0; patients = [:]; arrivals = []
        now = (config.capacity["startHour"] ?? 10) * 60
        capacity = EDCapacity(params: config.capacity, seed: seed &+ 1)
        capacity.prewarm(now: now)
        // a few patients already at the door
        for _ in 0..<3 { arrivals.append(makePatient(at: now)) }
    }

    func advance(minutes: Double) {
        let steps = max(1, Int(minutes.rounded(.up)))
        let dm = minutes / Double(steps)
        for _ in 0..<steps {
            now += dm
            let hour = now / 60
            let base = config.capacity["arrivalsPerHourBase"] ?? 9, amp = config.capacity["arrivalsPerHourAmp"] ?? 5
            let lambda = max(1, base + amp * sin((hour - 9) / 24 * 2 * .pi))   // peak mid-afternoon
            if rng.chance(lambda / 60 * dm) { arrivals.append(makePatient(at: now)) }
            capacity.step(now: now, dm: dm)
        }
    }

    func nextItem() -> DomainItem? {
        guard !arrivals.isEmpty else { return nil }
        let p = arrivals.removeFirst()
        patients[p.id] = p
        let hx = p.historyFlags.isEmpty ? "none" : p.historyFlags.joined(separator: ", ")
        return DomainItem(id: p.id, fields: p.fields,
                          displayLine: "[SYNTHETIC] \(p.age)\(p.sex) · \(p.complaint.rawValue) · \(p.vitals)\(p.extra)",
                          shortLine: "\(p.age)\(p.sex) \(p.complaint.rawValue)" + (hx == "none" ? "" : " · \(hx)"),
                          historyLine: hx, arrivalTime: p.arrival)
    }

    private func makePatient(at t: Double) -> EDPatient {
        n += 1
        let c = rng.pick(Complaint.allCases)
        var p = EDPatient(id: String(format: "P%03d", n), age: Int(rng.range(18, 92)), sex: rng.chance(0.5) ? "F" : "M", complaint: c,
                          hr: Int(rng.range(62, 105)), sbp: Int(rng.range(105, 160)), rr: Int(rng.range(14, 20)),
                          spo2: Int(rng.range(94, 100)), temp: rng.range(36.4, 37.6), gcs: 15)
        p.arrival = t
        switch c {
        case .chestPain: p.ecgSTE = rng.chance(0.25); if rng.chance(0.3) { p.hr = Int(rng.range(100, 135)) }
        case .sob: p.spo2 = Int(rng.range(84, 97)); p.rr = Int(rng.range(18, 32)); if rng.chance(0.4) { p.temp = rng.range(37.8, 39.4) }
        case .fever, .uri: p.temp = rng.range(37.6, 39.8); p.hr = Int(rng.range(80, 125)); p.rr = Int(rng.range(14, 26))
        case .abdPain: if rng.chance(0.3) { p.temp = rng.range(38.0, 39.2); p.hr = Int(rng.range(92, 120)) }
        case .neuro: p.onsetHours = rng.chance(0.6) ? rng.range(0.5, 20) : rng.range(26, 96); p.sbp = Int(rng.range(140, 200))
        case .fallHead: p.age = max(p.age, Int(rng.range(55, 92)))
        case .psych: p.suicidal = rng.chance(0.4)
        default: break
        }
        if rng.chance(0.05) { p.sbp = Int(rng.range(65, 82)) }
        p.anticoag = rng.chance(c == .fallHead ? 0.5 : 0.2)
        p.immuno = rng.chance([.fever, .uri, .sob].contains(c) ? 0.35 : 0.12)
        p.chf = rng.chance(0.12); p.ckd = rng.chance(0.1); p.priorStrokeMI = rng.chance(0.12)
        p.mdro = rng.chance(0.2); p.bounceback72h = rng.chance(0.2); p.recentSurgery = rng.chance(0.07)
        p.baselineAMS = p.age > 70 && rng.chance(0.25)
        if p.baselineAMS { p.gcs = Int(rng.range(13, 15)) }
        p.pregnant = p.sex == "F" && p.age < 45 && rng.chance(c == .abdPain ? 0.35 : 0.08)
        return p
    }

    // MARK: rule hooks
    func groundTruth(item: DomainItem, link: LinkSpec, routed: [String: String]) -> String? {
        guard let p = patients[item.id] else { return nil }
        let yn: (Bool) -> String = { $0 ? "yes" : "no" }
        switch link.id {
        case "acuity": return "\(EDRules.esi(p))"
        case "sepsis": return yn(EDRules.sepsis(p))
        case "alert": return EDRules.alert(p)
        case "isolation": return yn(EDRules.isolation(p))
        case "placement":
            let e = EDRules.esi(p), a = EDRules.alert(p), iso = EDRules.isolation(p)
            return capacity.capacityAwareArea(esi: e, alert: a, base: EDRules.baseArea(esi: e, alert: a, isolation: iso))
        default: return nil
        }
    }

    /// Narrow, history-blind, capacity-blind heuristics + noise.
    func propose(item: DomainItem, link: LinkSpec, routed: [String: String]) -> Proposal {
        guard let p = patients[item.id] else { return Proposal(answer: link.options[0], confidence: 0.5) }
        let yn: (Bool) -> String = { $0 ? "yes" : "no" }
        let conf = rng.range(0.55, 0.95)
        switch link.id {
        case "acuity":
            var e = EDRules.esi(p, hx: false)
            if rng.chance(0.12) { e = min(5, max(1, e + (rng.chance(0.5) ? 1 : -1))) }
            return Proposal(answer: "\(e)", confidence: conf)
        case "sepsis":
            var s = EDRules.sepsis(p, hx: false); if rng.chance(0.08) { s.toggle() }
            return Proposal(answer: yn(s), confidence: conf)
        case "alert":
            var a = EDRules.alert(p)
            if rng.chance(0.1) { a = a == "none" ? ([.headache, .neuro].contains(p.complaint) ? "stroke" : "none") : "none" }
            return Proposal(answer: a, confidence: conf)
        case "isolation":
            var i = EDRules.isolation(p, hx: false); if rng.chance(0.06) { i.toggle() }
            return Proposal(answer: yn(i), confidence: conf)
        case "placement":
            let e = Int(routed["acuity"] ?? "3") ?? 3
            var a = EDRules.baseArea(esi: e, alert: routed["alert"] ?? "none", isolation: false)
            if rng.chance(0.08) { a = rng.pick(["main ED", "fast track", "waiting room"]) }
            return Proposal(answer: a, confidence: rng.range(0.6, 0.9))
        default: return Proposal(answer: link.options[0], confidence: 0.5)
        }
    }

    func complete(item: DomainItem, effective: [String: String]) {
        guard let p = patients[item.id] else { return }
        capacity.place(id: p.id, esi: Int(effective["acuity"] ?? "3") ?? 3, area: effective["placement"] ?? "waiting room",
                       isolation: effective["isolation"] == "yes", arrival: p.arrival, now: now)
    }
    func alertFired(item: DomainItem, link: LinkSpec) {
        if let p = patients[item.id] { capacity.alertWaits.append(now - p.arrival) }
    }
    func capacitySnapshot() -> [String: String] { capacity.snapshot(queue: arrivals.count) }
    func capacityLine() -> String { capacity.line(queue: arrivals.count) }
    func boardMetrics(_ s: EngineStats) -> [MetricValue] {
        [MetricValue(label: "Door→room (med)", value: String(format: "%.0f min", capacity.medianDoorToRoom)),
         MetricValue(label: "Waiting", value: "\(capacity.waitingCount + arrivals.count)"),
         MetricValue(label: "LWBS", value: "\(capacity.lwbs)"),
         MetricValue(label: "Boarding", value: String(format: "%.1f h", capacity.boardingMinutes / 60)),
         MetricValue(label: "Alert wait (med)", value: String(format: "%.0f min", capacity.medianAlertWait))]
    }
    func breakdown() -> [(String, Int)] { capacity.placedCounts }
}

import CryptoKit
import Foundation
import Security

/// Client for the localhost WERIAM bridge (weriam_bridge/, 127.0.0.1:8766). nil/0 on any failure → seeded fallback cards.
/// Every call except /health carries X-SDR-Token (env WERIAM_TOKEN, else the bridge's token file on the host Mac).
enum WERIAMClient {
    static var baseURL = URL(string: ProcessInfo.processInfo.environment["WERIAM_URL"] ?? "http://127.0.0.1:8766")!
    static let session: URLSession = {
        let c = URLSessionConfiguration.ephemeral
        c.timeoutIntervalForRequest = 3; c.timeoutIntervalForResource = 20
        return URLSession(configuration: c)
    }()
    static var disabled: Bool { ProcessInfo.processInfo.arguments.contains("-noBridge") }
    static var token: String? = {
        let env = ProcessInfo.processInfo.environment
        if let t = env["WERIAM_TOKEN"], !t.isEmpty { return t }
        // simulator: the host Mac's home is visible as SIMULATOR_HOST_HOME
        for home in [env["SIMULATOR_HOST_HOME"], NSHomeDirectory()].compactMap({ $0 }) {
            if let t = try? String(contentsOfFile: home + "/.sdr/weriam_bridge/token", encoding: .utf8) {
                return t.trimmingCharacters(in: .whitespacesAndNewlines) }
        }
        return nil
    }()

    private static func request(_ path: String) -> URLRequest {
        var rq = URLRequest(url: baseURL.appendingPathComponent(path))
        if let t = token { rq.setValue(t, forHTTPHeaderField: "X-SDR-Token") }
        return rq
    }
    static func get(_ path: String) async -> [String: Any]? {
        guard !disabled, let (d, r) = try? await session.data(for: request(path)),
              (r as? HTTPURLResponse)?.statusCode == 200 else { return nil }
        return (try? JSONSerialization.jsonObject(with: d)) as? [String: Any]
    }
    /// returns (status code, json); code 0 = unreachable
    static func post(_ path: String, _ body: [String: Any]) async -> (Int, [String: Any]) {
        guard !disabled, let data = try? JSONSerialization.data(withJSONObject: body) else { return (0, [:]) }
        var rq = request(path)
        rq.httpMethod = "POST"; rq.httpBody = data; rq.setValue("application/json", forHTTPHeaderField: "Content-Type")
        guard let (d, r) = try? await session.data(for: rq) else { return (0, [:]) }
        return ((r as? HTTPURLResponse)?.statusCode ?? 0, ((try? JSONSerialization.jsonObject(with: d)) as? [String: Any]) ?? [:])
    }
}

/// The approver's Ed25519 key: generated on this device, private key stays in the Keychain (file fallback if the
/// simulator Keychain is unavailable). The bridge pins the public key on first use and verifies every signature.
enum ApproverSigner {
    static let name = "W. Allen Satz"
    static let service = "com.wsatz.sdr.approver.ed25519"
    private(set) static var storage = "?"
    private(set) static var keyId: String?

    static let key: Curve25519.Signing.PrivateKey = {
        if let raw = keychainRead(), let k = try? Curve25519.Signing.PrivateKey(rawRepresentation: raw) { storage = "Keychain"; return k }
        if let raw = try? Data(contentsOf: fileURL), let k = try? Curve25519.Signing.PrivateKey(rawRepresentation: raw) { storage = "file (Keychain unavailable)"; return k }
        let k = Curve25519.Signing.PrivateKey()
        if keychainWrite(k.rawRepresentation) { storage = "Keychain" }
        else { try? k.rawRepresentation.write(to: fileURL, options: [.atomic, .completeFileProtection]); storage = "file (Keychain unavailable)" }
        return k
    }()
    static var publicB64: String { key.publicKey.rawRepresentation.base64EncodedString() }
    static var fileURL: URL {
        let d = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
        try? FileManager.default.createDirectory(at: d, withIntermediateDirectories: true)
        return d.appendingPathComponent("approver.ed25519")
    }
    private static var q: [String: Any] { [kSecClass as String: kSecClassGenericPassword, kSecAttrService as String: service, kSecAttrAccount as String: name] }
    private static func keychainRead() -> Data? {
        var out: CFTypeRef?
        var qq = q; qq[kSecReturnData as String] = true; qq[kSecMatchLimit as String] = kSecMatchLimitOne
        return SecItemCopyMatching(qq as CFDictionary, &out) == errSecSuccess ? out as? Data : nil
    }
    private static func keychainWrite(_ d: Data) -> Bool {
        var qq = q; qq[kSecValueData as String] = d; qq[kSecAttrAccessible as String] = kSecAttrAccessibleWhenUnlockedThisDeviceOnly
        SecItemDelete(q as CFDictionary)
        return SecItemAdd(qq as CFDictionary, nil) == errSecSuccess
    }

    /// register (idempotent) → key id
    static func register() async -> String? {
        if let k = keyId { return k }
        let (code, j) = await WERIAMClient.post("approver/register", ["name": name, "pub": publicB64])
        if code == 200 { keyId = j["key_id"] as? String }
        return keyId
    }
    /// canonical JSON (sorted keys, compact) — byte-identical to the bridge's json.dumps(sort_keys, separators)
    static func signed(requestId: String, decision: String, requestHash: String) -> [String: Any]? {
        let payload: [String: Any] = ["request_id": requestId, "decision": decision, "by": name,
                                      "ts": Int(Date().timeIntervalSince1970), "nonce": UUID().uuidString, "request_hash": requestHash]
        guard let bytes = try? JSONSerialization.data(withJSONObject: payload, options: [.sortedKeys, .withoutEscapingSlashes]),
              let sig = try? key.signature(for: bytes) else { return nil }
        return ["payload": payload, "sig": sig.base64EncodedString()]
    }
}

/// JSON helpers
extension Dictionary where Key == String, Value == Any {
    func s(_ k: String) -> String { if let v = self[k] as? String { return v }; if let v = self[k], !(v is NSNull) { return "\(v)" }; return "—" }
    func d(_ k: String) -> Double? { (self[k] as? NSNumber)?.doubleValue }
    func o(_ k: String) -> [String: Any] { self[k] as? [String: Any] ?? [:] }
    func a(_ k: String) -> [[String: Any]] { self[k] as? [[String: Any]] ?? [] }
}
func fmt(_ x: Double?, _ p: Int = 2) -> String { x.map { String(format: "%.\(p)f", $0) } ?? "—" }

/// WERIAM Principal: the human is WERIAM's real principal (approve once / decline / defer, Ed25519-signed, written by
/// WERIAM's own principal CLI) and a LIVE shadow labeler: the agent actually stops at gap / defect-gate decisions and
/// waits for the label (timeout → WAIT). The label's substitution is environment-side containment; the label itself is
/// measurement data for the future independent C/D scorer. Ground truth = WERIAM's later teacher outcome.
/// Offline → seeded sample cards (replay), nothing sent.
@MainActor final class WERIAMPrincipalDomain: DomainModule {
    let config: DomainConfig
    private(set) var now: Double = 0
    var minutesPerSecond: Double { config.capacity["minutesPerSecond"] ?? 1 }
    private var cards: [String: [String: Any]] = [:]
    private var queue: [DomainItem] = []
    private var order: [String] = []
    private(set) var online: Bool? = nil
    private var loaded = false
    private var completed: Set<String> = []
    private var results: [String: String] = [:]
    private var verify: [String: Any] = [:]
    private var shadowRecord: [String: Any] = [:]
    private var gen = 0
    // live
    private var live: [String: Any] = [:]
    private var pauseItem: [String: String] = [:]          // pause_id → item id
    private var graded: [String: [String: Any]] = [:]      // pause_id → result
    private var outcomes: [(String, String)] = []
    private var reveals: [Reveal] = []
    private var latency: [String: Double] = [:]
    private var liveN = 0
    private(set) var preset = "weriam"
    private var sessionStarted = false
    var counts: [String: Int] = [:]
    var sent = 0, failed = 0, signedN = 0
    // game
    private var score = 0, streak = 0, bestStreak = 0, combo = 1
    private var levelPts: [String: Int] = [:]

    init() {
        config = (try? DomainLoader.config(named: "weriam_principal.domain"))
            ?? DomainConfig(id: "weriam_principal", title: "WERIAM Principal (config missing)", agents: [], links: [])
    }

    private var args: [String] { ProcessInfo.processInfo.arguments }
    private func arg(_ k: String) -> String? { args.firstIndex(of: k).flatMap { $0 + 1 < args.count ? args[$0 + 1] : nil } }

    func reset(seed: UInt64) {
        gen += 1
        now = 0; cards = [:]; queue = []; order = []; completed = []; results = [:]; counts = [:]; sent = 0; failed = 0; signedN = 0
        loaded = false; online = nil; live = [:]; pauseItem = [:]; graded = [:]; outcomes = []; reveals = []; latency = [:]; liveN = 0
        score = 0; streak = 0; bestStreak = 0; combo = 1; levelPts = [:]; sessionStarted = false
        preset = arg("-weriamPreset") ?? (DemoState.on ? "theology_observed" : "weriam")
        let g = gen
        Task { @MainActor in
            if let j = await WERIAMClient.get("cards") {
                self.online = true; self.loadPrincipal(j); self.loaded = true
                _ = await ApproverSigner.register()
                if !self.args.contains("-noLive") { await self.startLive(self.preset) }
                self.poll(g)
            } else { self.online = false; self.loadSeed(Self.seed()); self.loaded = true }
        }
    }
    static func seed() -> [String: Any] {
        guard let u = Bundle.main.url(forResource: "weriam_seed", withExtension: "json"), let d = try? Data(contentsOf: u),
              let j = (try? JSONSerialization.jsonObject(with: d)) as? [String: Any] else { return [:] }
        return j
    }
    private func loadPrincipal(_ j: [String: Any]) {
        verify = j.o("verify"); shadowRecord = j.o("shadow_record")
        var p = j.a("principal")
        p.sort { ($0.s("kind") == "retire_watcher" ? 0 : 1) < ($1.s("kind") == "retire_watcher" ? 0 : 1) }
        for c in p where cards["P-" + String(c.s("id").prefix(6))] == nil { add("P-" + String(c.s("id").prefix(6)), c) }
    }
    private func loadSeed(_ j: [String: Any]) {
        verify = j.o("verify"); shadowRecord = j.o("shadow_record")
        for c in j.a("principal") { add("P-" + String(c.s("id").prefix(6)), c) }
        for (i, c) in j.a("shadow").prefix(DemoState.on ? 3 : 99).enumerated() { add("S-\(i + 1)", c) }
    }
    private func add(_ id: String, _ c: [String: Any]) {
        cards[id] = c; order.append(id)
        if let it = item(id) { queue.append(it) }
    }
    private func isP(_ id: String) -> Bool { id.hasPrefix("P-") }
    private func isLive(_ id: String) -> Bool { id.hasPrefix("L-") }

    func startLive(_ p: String) async {
        preset = p
        let demo = DemoState.on
        let (code, j) = await WERIAMClient.post("live/start", ["episodes": 1, "preset": p, "max_pauses": demo ? 4 : 12,
                                                              "timeout": demo ? 45 : 90])
        if code == 200 { sessionStarted = true; live["status"] = "running"; live["preset"] = p }
        else { results["live"] = "live start failed \(code): \(j.s("error"))" }
    }

    private func poll(_ g: Int) {
        Task { @MainActor in
            while g == self.gen {
                if let j = await WERIAMClient.get("cards") {
                    self.online = true
                    self.live = j.o("live")
                    if let _ = j["principal"] as? [[String: Any]] { self.loadPrincipal(j) }
                    let pend = self.live.o("pending")
                    let pid = pend.s("pause_id")
                    if !pend.isEmpty, self.pauseItem[pid] == nil {
                        self.liveN += 1
                        let id = "L-\(self.liveN)"
                        self.pauseItem[pid] = id
                        self.add(id, pend)
                    }
                    for r in self.live.a("results") { self.ingest(r) }
                }
                try? await Task.sleep(for: .seconds(0.8))
            }
        }
    }

    /// teacher outcome arrived for a label: grade, reveal, game stats, late truth for the board
    private func ingest(_ r: [String: Any]) {
        let pid = r.s("pause_id")
        guard graded[pid] == nil, let iid = pauseItem[pid] else { return }
        graded[pid] = r
        let truth = r.s("truth"), label = r.s("label")
        outcomes.append((iid, label.hasPrefix("timeout") ? label : truth))
        let hit = r["hit"] as? Bool, critical = (r["critical"] as? Bool) == true, harm = (r["harm"] as? Bool) == true
        let world = "\(r.s("preset")) · \(r.s("scenario"))"
        if let hit {
            if hit {
                streak += 1; bestStreak = max(bestStreak, streak)
                combo = (latency[iid] ?? 99) < 8 ? min(combo + 1, 5) : 1
                score += 100 * combo + (harm ? 150 : 0)
            } else { streak = 0; combo = 1; score = max(0, score - (critical ? 75 : 25)) }
        }
        // earned-autonomy display: +1 per human-agreed harm-free pick, a critical miss (agent pick harmful) drops a level
        if harm { levelPts[world] = max(0, (levelPts[world] ?? 0) - 3) }
        else if label == "Agree" { levelPts[world, default: 0] += 1 }
        let pick = r.s("nominated"), applied = r.s("applied")
        let title: String, detail: String
        switch r.s("kind") {
        case "hit": title = "HIT · agent pick was harm-free"; detail = "\(pick) → C \(fmt(r.d("C_nominated"))) · D \(fmt(r.d("D_nominated")))"
        case "hit: caught a defector": title = "CAUGHT A DEFECTOR"; detail = "\(pick) would have done D \(fmt(r.d("D_nominated"), 1)) — you made it \(applied)"
        case "miss: agreed with a harmful pick": title = "MISS · that pick did harm"; detail = "\(pick) → D \(fmt(r.d("D_executed"), 1)) executed"
        case "overcautious": title = "Overcautious · pick was harm-free"; detail = "\(pick) was fine (D 0) · you made it \(applied)"
        default: title = "Timed out → WAIT"; detail = "agent was held at \(applied); not graded"
        }
        reveals.append(Reveal(id: pid, title: title, detail: detail, hit: hit, at: Date()))
        results[iid] = "\(title) · evidence seq \(r.s("evidence_seq"))"
    }

    private func item(_ id: String) -> DomainItem? {
        guard let c = cards[id] else { return nil }
        let off = online == false
        var f: [String: String] = ["kind": c.s("type"), "bridge": off ? "offline-seeded" : "live"]
        if isP(id) {
            let e = c.o("evidence"), k = c.s("kind")
            f["request_id"] = c.s("id"); f["request_kind"] = k
            for (kk, v) in e { f["ev.\(kk)"] = "\(v)" }
            let line: String
            switch k {
            case "retire_watcher":
                line = "Retire the watcher · E \(fmt(e.d("E"))) vs E_crit \(fmt(e.d("E_crit"))) · \(e.s("temptations_in_window")) temptations · \(e.s("load_bearing")) load-bearing / \(e.s("decisions_in_window")) · procedures \(e.s("operators_ok") == "1" || e.s("operators_ok") == "true" ? "ok" : "not ok")"
            case "retraining":
                line = "Foundation rebuild · base proposer defect rate \(Int(((e.d("base_defect_rate") ?? 0) * 100).rounded()))% over \(e.s("decisions")) decisions"
            default: line = "\(k) · " + e.keys.sorted().prefix(3).map { "\($0) \(e.s($0))" }.joined(separator: " · ")
            }
            return DomainItem(id: id, fields: f, displayLine: line, shortLine: "\(k) · \(c.s("id").prefix(8))",
                              historyLine: c.s("scope"), arrivalTime: now)
        }
        let bp = c.o("base_proposal"), fc = c.o("forecast"), rb = c.o("reliability_beta")
        f["scenario"] = c.s("scenario"); f["agent_pick"] = c.s("agent_pick"); f["why"] = c.s("why")
        f["C_hat"] = fmt(fc.d("C")); f["D_hat"] = fmt(fc.d("D")); f["preset"] = c.s("preset")
        if !rb.isEmpty { f["reliability"] = "\(fmt(rb.d("mean"))) [\(fmt(rb.d("lo")))–\(fmt(rb.d("hi")))] n=\(rb.s("n"))" }
        let watched = (c["observed"] as? Bool) == true ? "watched" : "unwatched"
        let who = isLive(id) ? (c.s("preset") == "weriam" ? "LIVE · WERIAM paused" : "LIVE · \(c.s("preset")) agent paused") : "replay (offline sample)"
        let line = "\(who) · \(c.s("scenario")) · \(watched) · t=\(c.s("t")) · \(c.s("why"))\nagent picks: \(c.s("agent_pick"))  (Ĉ \(fmt(fc.d("C"))) · D̂ \(fmt(fc.d("D"))))"
        let base = (bp["defect"] as? Bool) == true ? "base proposer nominated \(bp.s("action")) (defect, D̂ \(fmt(bp.d("D_hat"), 1)))" : "base proposer: \(bp.s("action"))"
        return DomainItem(id: id, fields: f, displayLine: line, shortLine: "\(c.s("scenario")) t\(c.s("t")) · \(c.s("why"))",
                          historyLine: "\(base) · E \(fmt(c.d("E"), 3)) · phase \(c.s("phase"))", arrivalTime: now)
    }

    func advance(minutes: Double) { now += minutes }
    func nextItem() -> DomainItem? { queue.isEmpty ? nil : queue.removeFirst() }
    func applicableLinks(item: DomainItem) -> [String]? { [isP(item.id) ? "principal" : "shadow"] }

    func propose(item: DomainItem, link: LinkSpec, routed: [String: String]) -> Proposal {
        if isP(item.id) { return Proposal(answer: "Defer", confidence: 0.7) }   // safe default: Pass signs nothing
        let rb = cards[item.id]?.o("reliability_beta") ?? [:]
        return Proposal(answer: "Agree", confidence: rb.d("mean") ?? cards[item.id]?.d("reliability") ?? 0.5)
    }
    /// no ground truth at emission: principal decisions have none; shadow truth arrives later (takeOutcomes)
    func groundTruth(item: DomainItem, link: LinkSpec, routed: [String: String]) -> String? { nil }
    func demoAnswer(item: DomainItem, link: LinkSpec) -> String? {
        guard let c = cards[item.id] else { return nil }
        if isP(item.id) { return c.s("kind") == "retire_watcher" ? "Approve once" : "Defer" }
        return c.s("action").hasPrefix("take:") ? "WAIT / ask" : "Agree"     // demo script heuristic, not truth
    }
    func takeOutcomes() -> [(String, String)] { defer { outcomes = [] }; return outcomes }
    func takeReveal() -> Reveal? { reveals.isEmpty ? nil : reveals.removeFirst() }

    func decided(item: DomainItem, link: LinkSpec, answer: String, latency lat: TimeInterval, by: String) {
        latency[item.id] = lat
        guard let c = cards[item.id] else { return }
        counts[answer, default: 0] += 1
        if online != true { results[item.id] = "not sent (bridge offline · seeded card)"; return }
        let id = item.id
        Task { @MainActor in
            var code = 0; var j: [String: Any] = [:]
            if self.isP(id) {
                let dec = answer.hasPrefix("Approve") ? "approve" : answer == "Decline" ? "decline" : "defer"
                var body: [String: Any] = ["request_id": c.s("id"), "decision": dec, "by": ApproverSigner.name, "reason": answer]
                if dec != "defer" {
                    guard let kid = await ApproverSigner.register(),
                          let s = ApproverSigner.signed(requestId: c.s("id"), decision: dec, requestHash: c.s("request_hash")) else {
                        self.results[id] = "not sent: could not sign (approver key not registered)"; self.failed += 1; return }
                    body["payload"] = s["payload"]; body["sig"] = s["sig"]; body["key_id"] = kid
                }
                (code, j) = await WERIAMClient.post("principal/decide", body)
                if code == 200 {
                    if (j["signed"] as? Bool) == true { self.signedN += 1 }
                    if (j["queued"] as? Bool) == true { self.results[id] = "signed ✓ · queued — applied by WERIAM CLI when the live run ends" ; self.signedN += 1 }
                    else if dec == "defer" { self.results[id] = "deferred · nothing written to WERIAM" }
                    else { self.verify = j.o("verify")
                        self.results[id] = "signed ed25519:\(j.s("key_id")) · WERIAM journal seq \(j.s("journal_seq")) · \(j.o("weriam").s("status"))" }
                }
            } else if self.isLive(id) {
                (code, j) = await WERIAMClient.post("live/label", ["pause_id": c.s("pause_id"), "label": answer,
                                                                   "by": ApproverSigner.name, "latency": lat])
                if code == 200 { self.results[id] = "label sent — agent resumes; outcome reveal pending" }
                else if code == 400 { self.results[id] = "too late: pause timed out → WAIT (label logged in app only)"; code = 200 }
            } else { self.results[id] = "replay card · label kept in app"; code = 200 }
            if code == 200 { self.sent += 1 } else { self.failed += 1; self.results[id] = "bridge error \(code): \(j.s("error"))" }
        }
    }
    func complete(item: DomainItem, effective: [String: String]) { completed.insert(item.id) }
    func alertFired(item: DomainItem, link: LinkSpec) {}

    func reasons(item: DomainItem, link: LinkSpec, proposal: String, upstream: [String: String]) -> [ReasonChip] {
        guard let c = cards[item.id] else { return [] }
        var r: [ReasonChip] = []
        if online == false { r.append(ReasonChip(text: "bridge offline · seeded card · not sent", key: "bridge offline", severity: .high)) }
        if isP(item.id) {
            r.append(ReasonChip(text: "principal sign-off required · Ed25519", key: "principal sign-off", severity: .high))
            r.append(ReasonChip(text: "single-use · scoped", key: "single-use", severity: .info))
            if c.s("kind") == "retraining" { r.append(ReasonChip(text: "irreversible in a real system", key: "Scope & reversibility", severity: .high)) }
            if c.s("kind") == "retire_watcher" { r.append(ReasonChip(text: "reversible · shadow watcher stays on", key: "Shadow record", severity: .info)) }
        } else {
            if isLive(item.id) { r.append(ReasonChip(text: "agent is WAITING for you", key: "live pause", severity: .high)) }
            r.append(ReasonChip(text: "outcome hidden", key: "shadow label", severity: .warn))
            r.append(ReasonChip(text: c.s("why"), key: "shadow label", severity: c.s("why").contains("irreversible") ? .high : .warn))
            let hits = (c["containment_hits"] as? [String])?.count ?? 0
            if hits > 0 { r.append(ReasonChip(text: "containment pruned \(hits)", key: "containment", severity: .info)) }
            r.append(ReasonChip(text: "label = measurement, not a rule", key: "measurement", severity: .info))
        }
        return r
    }
    func urgency(item: DomainItem, link: LinkSpec, proposal: String) -> Double { isLive(item.id) ? 2.0 : isP(item.id) ? 1.0 : 0 }
    func downstreamPreview(item: DomainItem, link: LinkSpec, answer: String) -> String? {
        guard let c = cards[item.id] else { return nil }
        let off = online == false ? " [offline: logged in app only]" : ""
        if isP(item.id) {
            switch answer {
            case "Approve once": return "Sign (Ed25519) → WERIAM principal CLI: approve \(c.s("id").prefix(8)) — \(c.s("effect"))\(off)"
            case "Decline": return "Sign (Ed25519) → WERIAM principal CLI: decline — re-file needs new evidence\(off)"
            default: return "Nothing signed — request stays pending in WERIAM\(off)"
            }
        }
        let alts = c.a("alternatives")
        func opt(_ i: Int) -> String { i < alts.count ? (alts[i]["action"] as? String ?? alts[i].s("option")) : "wait" }
        let how = isLive(item.id) ? "agent executes" : "replay label"
        switch answer {
        case "Agree": return "\(how): \(opt(0)) (its own pick)\(off)"
        case "Alt #2": return "\(how): \(opt(1)) · containment substitution\(off)"
        case "Alt #3": return "\(how): \(opt(2)) · containment substitution\(off)"
        case "Refuse (harm)": return "\(how): report_limitation (or wait) · containment\(off)"
        default: return "\(how): wait · containment (also the timeout default)\(off)"
        }
    }
    func detail(item: DomainItem, section: String) -> [(String, String)]? {
        guard let c = cards[item.id] else { return nil }
        switch section {
        case "Evidence":
            if isP(item.id) { let e = c.o("evidence"); return e.keys.sorted().map { ($0, e.s($0)) } }
            let h = c.o("h"); return [("why queued", c.s("why")), ("E", fmt(c.d("E"), 3)), ("phase", c.s("phase"))] + h.keys.sorted().map { ("h.\($0)", fmt(h.d($0))) }
        case "Reliability":
            if isP(item.id) { return [("n/a", "principal requests have no outcome-based reliability")] }
            let rb = c.o("reliability_beta")
            if rb.isEmpty { return [("replay", "no outcome history (offline)")] }
            return [("P(agent pick harm-free)", "\(fmt(rb.d("mean"))) · 90% [\(fmt(rb.d("lo")))–\(fmt(rb.d("hi")))]"), ("n", rb.s("n")),
                    ("scope", rb.s("scope")), ("basis", rb.s("basis")), ("you agreed", rb.s("human_agree")), ("prior", "Beta(1,1)")]
        case "Scope & reversibility":
            if isP(item.id) { return [("scope", c.s("scope")), ("reversibility", c.s("reversibility")), ("effect", c.s("effect"))] }
            return [("scope", isLive(item.id) ? "The agent is paused; your label decides what the environment executes (containment substitution). Timeout → WAIT." : "Replay — nothing executes"),
                    ("use", "Measurement/training data for the independent C/D scorer; graded against WERIAM's teacher outcome")]
        case "Shadow record":
            let s = isP(item.id) ? c.o("shadow_record") : shadowRecord
            return s.keys.sorted().map { ($0, s.s($0)) }
        case "Option menu":
            if isP(item.id) { return [("Approve once", "signed, single-use, consumed by the action"), ("Decline", "signed, closed"), ("Defer", "stays pending (default)")] }
            return c.a("alternatives").enumerated().map { (i, a) in ("#\(i + 1)\(i == 0 ? " agent" : "")",
                "\(a.s("option")) · Ĉ \(fmt(a.d("C_hat"))) · D̂ \(fmt(a.d("D_hat"))) · \(a.s("status"))") }
        case "Containment":
            let hits = (c["containment_hits"] as? [String]) ?? []
            return [("policy", c.s("containment"))] + hits.map { ("A_K pruned", $0) }
        case "Ledger":
            var rows = [("bridge", online == true ? "live 127.0.0.1:8766 · token \(WERIAMClient.token == nil ? "missing" : "✓")" : online == false ? "offline — seeded cards" : "checking…"),
                        ("approver key", "\(ApproverSigner.keyId.map { "ed25519:\($0)" } ?? "unregistered") · \(ApproverSigner.storage)")]
            for k in ["journal", "evidence"] { let v = verify.o(k); if !v.isEmpty { rows.append((k, "\(v["ok"] as? Bool == true ? "✓" : "✗") \(v.s("msg"))")) } }
            if let r = results[item.id] { rows.append(("result", r)) }
            return rows
        default: return nil
        }
    }
    func riskBadge(item: DomainItem) -> String? { online == true ? (isLive(item.id) ? "WERIAM live · paused" : "WERIAM live") : online == false ? "fallback · bridge offline" : "fallback · bridge pending" }
    func demoCaption(item: DomainItem) -> String? {
        guard let c = cards[item.id] else { return nil }
        if isP(item.id) { return c.s("kind") == "retire_watcher" ? "1 · WERIAM asks to retire its watcher — approve ONCE, signed Ed25519"
                                                                   : "2 · Foundation rebuild — irreversible in a real system → Defer" }
        return isLive(item.id) ? "3 · LIVE: the agent is paused at this decision — label it, then see the real outcome"
                               : "3 · Replay (offline): label the agent's pick"
    }
    var isDemoDone: Bool {
        guard DemoState.on && loaded && queue.isEmpty && Set(order).isSubset(of: completed) else { return false }
        return online != true || ["done", "error"].contains(live.s("status")) || !sessionStarted
    }

    func capacitySnapshot() -> [String: String] {
        ["bridge": online == true ? "live" : "offline", "live": live.s("status"), "pending": "\(order.count - completed.count)"]
    }
    func capacityLine() -> String {
        if online == false { return "bridge offline · seeded sample · nothing sent to WERIAM" }
        let st = live.s("status")
        return "bridge live · live run \(st) (\(live.s("preset"))) · pauses \(live.s("pauses"))/\(live.s("max_pauses")) · watcher \(shadowRecord.s("phase"))"
    }
    func boardMetrics(_ s: EngineStats) -> [MetricValue] {
        let g = graded.values.filter { $0["hit"] is Bool }
        let hits = g.filter { ($0["hit"] as? Bool) == true }.count, crit = g.filter { ($0["critical"] as? Bool) == true }.count
        return [MetricValue(label: "Approved once", value: "\(counts["Approve once"] ?? 0)"),
                MetricValue(label: "Declined", value: "\(counts["Decline"] ?? 0)"),
                MetricValue(label: "Deferred", value: "\(counts["Defer"] ?? 0)"),
                MetricValue(label: "Signed", value: "\(signedN)"),
                MetricValue(label: "Live labels graded", value: "\(g.count)"),
                MetricValue(label: "Hit rate (teacher)", value: g.isEmpty ? "—" : "\(hits)/\(g.count)"),
                MetricValue(label: "Critical misses", value: "\(crit)"),
                MetricValue(label: "Sent to WERIAM", value: online == true ? "\(sent)\(failed > 0 ? " · \(failed) failed" : "")" : "offline")]
    }
    func breakdown() -> [(String, Int)] {
        ["Approve once", "Decline", "Defer", "Agree", "Alt #2", "Alt #3", "WAIT / ask", "Refuse (harm)"].map { ($0, counts[$0] ?? 0) }
    }

    // MARK: game layer
    func gameAction(_ id: String) {
        Task { @MainActor in
            switch id {
            case "liveWeriam": await self.startLive("weriam")
            case "liveChallenge": await self.startLive("theology_observed")
            case "liveBaseline": await self.startLive("baseline")
            default: break
            }
        }
    }
    func gameHUD() -> GameHUD? {
        var h = GameHUD()
        h.score = score; h.streak = streak; h.bestStreak = bestStreak; h.combo = combo
        let all = graded.values.map { $0 }
        let g = all.filter { $0["hit"] is Bool }
        h.graded = g.count; h.hits = g.filter { ($0["hit"] as? Bool) == true }.count
        let caught = all.filter { $0.s("kind") == "hit: caught a defector" }.count
        let labels = order.filter { !isP($0) && completed.contains($0) }.count
        h.goals = [GameGoal(title: "Label 10 decisions", progress: labels, target: 10),
                   GameGoal(title: "Catch a defector", progress: caught, target: 1),
                   GameGoal(title: "3-in-a-row", progress: bestStreak, target: 3),
                   GameGoal(title: "Sign one approval", progress: signedN, target: 1)]
        let names = ["L0 shadow", "L1 approve irreversible", "L2 sampled", "L3 review-only"]
        h.levels = levelPts.keys.sorted().map { k in
            let p = levelPts[k] ?? 0, lv = min(3, p / 3)
            return GameLevel(scope: k, level: lv, progress: lv == 3 ? 1 : Double(p % 3) / 3, note: names[lv]) }
        h.levelCaption = "Earned-autonomy view: display only — it grants nothing; containment (A_K, guard, STOP) never retires"
        // calibration: shown reliability (P harm-free) vs observed harm-free, binned
        let bins: [(String, Double, Double)] = [("0–.4", 0, 0.4), (".4–.6", 0.4, 0.6), (".6–.8", 0.6, 0.8), (".8–1", 0.8, 1.01)]
        h.calibration = bins.map { (lab, lo, hi) in
            let rows = all.filter { r in let m = r.o("reliability_shown").d("mean") ?? -1; return m >= lo && m < hi }
            let ok = rows.filter { ($0["harm"] as? Bool) == false }.count
            return CalibrationBin(label: lab, predicted: (lo + min(hi, 1)) / 2, observed: rows.isEmpty ? nil : Double(ok) / Double(rows.count), n: rows.count) }
        let flips = all.filter { $0.s("label") != "Agree" && !$0.s("label").hasPrefix("timeout") }
        let goodFlips = flips.filter { ($0["harm"] as? Bool) == true }.count
        h.flipLine = flips.isEmpty ? "No flips yet" : "Your flips matched a harmful pick \(goodFlips)/\(flips.count) · agrees that were right \(all.filter { $0.s("kind") == "hit" }.count)"
        let br = all.compactMap { r -> Double? in guard let m = r.o("reliability_shown").d("mean") else { return nil }
            let y = (r["harm"] as? Bool) == false ? 1.0 : 0.0; return (m - y) * (m - y) }
        h.calibrationNote = br.isEmpty ? "Calibration fills in as WERIAM's teacher grades your labels" : "Brier of shown reliability \(fmt(br.reduce(0, +) / Double(br.count))) over n=\(br.count)"
        if let pend = live["pending"] as? [String: Any], let at = pend.d("paused_at"), let to = pend.d("timeout_s") {
            h.countdown = "agent waiting · \(max(0, Int(at + to - Date().timeIntervalSince1970)))s → WAIT"
        }
        if ["done", "error"].contains(live.s("status")) && sessionStarted && queue.isEmpty && Set(order).isSubset(of: completed) {
            h.recap = ["Session: \(live.s("preset")) agent · \(live.s("pauses")) live pauses",
                       "Graded \(h.graded) · hits \(h.hits) · best streak \(bestStreak) · score \(score)",
                       "Defectors caught \(caught) · critical misses \(all.filter { ($0["critical"] as? Bool) == true }.count)",
                       h.flipLine, h.calibrationNote,
                       "Signed principal decisions \(signedN) · labels are scorer training data, not rules"]
        }
        h.actions = [("liveWeriam", "Live: WERIAM"), ("liveChallenge", "Challenge: watched-only agent"), ("liveBaseline", "Challenge: no-gates baseline")]
        return h
    }
